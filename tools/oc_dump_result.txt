import 'dart:async';
import 'dart:convert';
import 'dart:io';

import 'package:http/http.dart' as http;
import 'package:supabase_flutter/supabase_flutter.dart';

import '../../config.dart';
import '../cloud.dart';
import '../rewards.dart';

/// One HTTP answer from a billing function.
class BillingHttpResponse {
  final int status;
  final String body;
  final Map<String, String> headers;
  const BillingHttpResponse(this.status, this.body, [this.headers = const {}]);
}

/// The request never got an answer: [kind] is 'offline' (no connection) or 'timeout' (sent, no answer in time).
class BillingNetworkError implements Exception {
  final String kind;
  const BillingNetworkError(this.kind);
  @override
  String toString() => 'BillingNetworkError($kind)';
}

/// Everything the billing service needs from the outside world, so tests can replace it.
abstract class BillingApi {
  /// A user RPC (billing_summary, billing_history, billing_exchange, billing_buy_pass, billing_set_auto_exchange).
  Future<dynamic> rpc(String fn, {Map<String, dynamic>? params});

  /// POST billing-checkout with the attempt's idempotency key. Throws [BillingNetworkError] when there is no answer.
  Future<BillingHttpResponse> postCheckout(Map<String, dynamic> body, {required String idempotencyKey});

  /// GET billing-checkout?purchase=<id>. Throws [BillingNetworkError] when there is no answer.
  Future<BillingHttpResponse> getStatus(String purchaseId);

  /// Sends the billing-return request the payment page was redirected to (without following its redirect), so the
  /// server still binds the payment ref in that URL even though the in-app page stopped loading it.
  Future<void> touchReturn(Uri url);

  /// The Supabase project host (xyz.supabase.co): the only non-Sifalo host the payment page may load.
  String get functionsHost;

  /// The signed-in user (pending purchases are kept per user), or null.
  String? get userId;

  /// This install's id (first top-up bonus is once per device too).
  Future<String?> deviceId();
}

/// The real thing: Supabase RPCs + the billing-checkout edge function, signed with the user's session.
class SupabaseBillingApi implements BillingApi {
  SupabaseBillingApi({http.Client? client}) : _client = client ?? http.Client();
  final http.Client _client;

  static const postTimeout = Duration(seconds: 150); // the server waits up to 120 s on Sifalo
  static const getTimeout = Duration(seconds: 30);

  String get _checkout => '${AppConfig.supabaseUrl}/functions/v1/billing-checkout';

  @override
  String get functionsHost => Uri.parse(AppConfig.supabaseUrl).host;

  @override
  String? get userId {
    try {
      return Supabase.instance.client.auth.currentUser?.id;
    } catch (_) {
      return null;
    }
  }

  @override
  Future<String?> deviceId() async {
    try {
      return await Rewards.I.deviceId();
    } catch (_) {
      return null;
    }
  }

  @override
  Future<dynamic> rpc(String fn, {Map<String, dynamic>? params}) async {
    try {
      return await Supabase.instance.client.rpc(fn, params: params).timeout(getTimeout);
    } on TimeoutException {
      throw const BillingNetworkError('timeout');
    } on SocketException {
      throw const BillingNetworkError('offline');
    } on http.ClientException {
      throw const BillingNetworkError('offline');
    }
  }

  Future<Map<String, String>> _headers({String? idem}) async => {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer ${await Cloud.token()}',
        'apikey': AppConfig.supabaseAnonKey,
        if (idem != null) 'x-idempotency-key': idem,
      };

  Future<BillingHttpResponse> _send(Future<http.Response> Function() call, Duration timeout) async {
    try {
      final r = await call().timeout(timeout);
      return BillingHttpResponse(r.statusCode, r.body, r.headers);
    } on TimeoutException {
      throw const BillingNetworkError('timeout');
    } on SocketException {
      throw const BillingNetworkError('offline');
    } on http.ClientException {
      throw const BillingNetworkError('offline');
    } on HandshakeException {
      throw const BillingNetworkError('offline');
    }
  }

  @override
  Future<BillingHttpResponse> postCheckout(Map<String, dynamic> body, {required String idempotencyKey}) async {
    final h = await _headers(idem: idempotencyKey);
    return _send(() => _client.post(Uri.parse(_checkout), headers: h, body: jsonEncode(body)), postTimeout);
  }

  @override
  Future<BillingHttpResponse> getStatus(String purchaseId) async {
    final h = await _headers();
    return _send(() => _client.get(Uri.parse('$_checkout?purchase=${Uri.encodeQueryComponent(purchaseId)}'), headers: h), getTimeout);
  }

  @override
  Future<void> touchReturn(Uri url) async {
    try {
      final req = http.Request('GET', url)..followRedirects = false;
      final r = await _client.send(req).timeout(const Duration(seconds: 25));
      await r.stream.drain<void>().timeout(const Duration(seconds: 5));
    } catch (_) {
      // Best effort: Sifalo's webhook and the status poll confirm the payment too.
    }
  }
}

int _int(Object? v, [int d = 0]) => v is num ? v.toInt() : int.tryParse('${v ?? ''}') ?? d;
int? _intOrNull(Object? v) => v == null ? null : (v is num ? v.toInt() : int.tryParse('$v'));
double _dbl(Object? v, [double d = 0]) => v is num ? v.toDouble() : double.tryParse('${v ?? ''}') ?? d;
DateTime? _time(Object? v) => v == null ? null : DateTime.tryParse('$v')?.toUtc();
String? _str(Object? v) => v == null || '$v'.isEmpty ? null : '$v';
Map<String, dynamic> _map(Object? v) => v is Map ? v.map((k, x) => MapEntry('$k', x)) : const {};
List<Map<String, dynamic>> _maps(Object? v) => v is List ? [for (final x in v) if (x is Map) _map(x)] : const [];

/// Default tokens per reply (1M ≈ 270 replies) until the server says otherwise.
const kDefaultTokensPerReply = 3700;

/// How many replies [tokens] buys (rounded down, never negative).
int repliesFor(int tokens, {int perReply = kDefaultTokensPerReply}) {
  final per = perReply <= 0 ? kDefaultTokensPerReply : perReply;
  return tokens <= 0 ? 0 : (tokens / per).floor();
}

/// "About 270 replies". Main screens count replies, never "tokens" and never "unlimited".
String repliesText(int tokens, {int perReply = kDefaultTokensPerReply}) {
  final n = repliesFor(tokens, perReply: perReply);
  return tr(n == 1 ? 'About {n} reply' : 'About {n} replies', {'n': groupDigits(n)});
}

/// 1000000 -> "1,000,000".
String groupDigits(int n) {
  final s = n.abs().toString();
  final b = StringBuffer(n < 0 ? '-' : '');
  for (var i = 0; i < s.length; i++) {
    if (i > 0 && (s.length - i) % 3 == 0) b.write(',');
    b.write(s[i]);
  }
  return b.toString();
}

/// 1000000 -> "1M", 120000 -> "120k", 2500000 -> "2.5M". Used on pay-as-you-go only.
String tokensShort(int n) {
  String trim(double v) => v == v.roundToDouble() ? v.toInt().toString() : v.toStringAsFixed(1);
  if (n >= 1000000) return '${trim(n / 1000000)}M';
  if (n >= 1000) return '${trim(n / 1000)}k';
  return '$n';
}

/// "$4.99"
String usdText(double v) => '\$${v.toStringAsFixed(2)}';

/// "5 h 4 min", "12 min", "3 days".
String durationWords(Duration d) {
  final m = (d.inSeconds / 60).ceil().clamp(1, 1 << 30);
  if (m >= 48 * 60) return tr('{n} days', {'n': '${(m / 1440).round()}'});
  final h = m ~/ 60, mm = m % 60;
  if (h == 0) return tr('{n} min', {'n': '$mm'});
  return mm == 0 ? tr('{n} h', {'n': '$h'}) : tr('{h} h {m} min', {'h': '$h', 'm': '$mm'});
}

/// "00:00" on the phone's clock for the server's reset instant (midnight EAT). The app never assumes UTC.
String clockText(DateTime at) {
  final l = at.toLocal();
  return '${l.hour.toString().padLeft(2, '0')}:${l.minute.toString().padLeft(2, '0')}';
}

/// `billing_activity()` (payments + what was added or taken back), newest first.
List<HistoryItem> parseActivity(Object? v, {List<PlanInfo> catalog = const [], int perReply = kDefaultTokensPerReply}) {
  final j = _map(v);
  String nameOf(Object? id) {
    for (final p in catalog) {
      if (p.id == id) return p.label;
    }
    return id == null ? '' : planName('$id');
  }

  final out = <HistoryItem>[];
  for (final p in _maps(j['payments'])) {
    final st = '${p['status'] ?? ''}';
    final title = tr(st == 'paid'
        ? 'Paid'
        : st == 'pending'
            ? 'Payment being confirmed'
            : st == 'failed'
                ? 'Payment not completed'
                : st == 'reversed'
                    ? 'Payment refunded'
                    : 'Payment');
    out.add(HistoryItem(
      id: 'p${p['id']}',
      kind: 'payment_$st',
      title: title,
      detail: [nameOf(p['plan_id']), if (p['amount_usd'] != null) usdText(_dbl(p['amount_usd']))].where((s) => s.isNotEmpty).join(' · '),
      at: _time(p['created_at']),
    ));
  }
  for (final g in _maps(j['grants'])) {
    final kind = '${g['kind'] ?? ''}';
    if (kind == 'grant') continue; // the payment row already shows it
    final n = _int(g['tokens']);
    out.add(HistoryItem(
      id: 'g${g['id']}',
      kind: 'ledger_$kind',
      title: tr(n < 0 ? 'Taken back' : 'Added'),
      detail: repliesText(n.abs(), perReply: perReply),
      at: _time(g['created_at']),
    ));
  }
  out.sort((a, b) => (b.at ?? DateTime(0)).compareTo(a.at ?? DateTime(0)));
  return out;
}

/// A usage refusal from ai-chat (Contract A): limit_daily, limit_total, no_plan, plus busy (too_long, blocked and
/// server are not billing). Shapes: {error:{code,message}, billing?:{reason, resets_at, ...}} or {code, message, resets_at}.
@immutable
class BillingNotice {
  static const knownCodes = {'limit_daily', 'limit_total', 'no_plan', 'busy'};
  static const limitCodes = {'limit_daily', 'limit_total', 'no_plan'};

  final String code;
  final String reason;
  final String message;
  final int status;
  final DateTime? resetsAt;
  final int? retryAfterSeconds;
  final DateTime seenAt;

  const BillingNotice({
    required this.code,
    this.reason = '',
    this.message = '',
    this.status = 0,
    this.resetsAt,
    this.retryAfterSeconds,
    required this.seenAt,
  });

  /// null when [body] is not a usage refusal. Never throws, whatever the body.
  static BillingNotice? tryParse(int status, Object? body, {DateTime? now}) {
    try {
      final j = _map(body);
      final e = _map(j['error']);
      final bm = _map(j['billing']);
      final code = '${e['code'] ?? j['code'] ?? ''}';
      if (code.isEmpty) return null;
      // Without a billing block only the three limit codes count (a plain `busy` stays a normal error).
      if (bm.isEmpty && !limitCodes.contains(code)) return null;
      return BillingNotice(
        code: code,
        reason: '${bm['reason'] ?? e['reason'] ?? j['reason'] ?? ''}',
        message: '${e['message'] ?? j['message'] ?? ''}',
        status: status,
        resetsAt: _time(bm['resets_at'] ?? j['resets_at'] ?? e['resets_at']),
        retryAfterSeconds: _intOrNull(j['retry_after_seconds'] ?? e['retry_after']),
        seenAt: now ?? DateTime.now(),
      );
    } catch (_) {
      return null;
    }
  }

  bool get known => knownCodes.contains(code);

  /// Info only (waiting helps), versus a limit (getting more helps).
  bool get isInfo => code == 'busy';

  /// One short line, the sheet's title.
  String get title {
    switch (code) {
      case 'limit_daily':
        return tr(reason == 'bot_share' ? 'Bots have used their share for today.' : "You've used today's replies.");
      case 'limit_total':
        return tr("You've used all your replies.");
      case 'no_plan':
        return tr('You need a plan for this.');
      case 'busy':
        return tr('SubMac is busy. Try again soon.');
    }
    return tr("This reply couldn't be sent.");
  }

  /// "Resets at midnight." / "Resets at 00:00 (in 5 h 4 min)." from the time the server sent.
  String resetLine({DateTime? now}) {
    if (code != 'limit_daily') return '';
    final at = resetsAt;
    if (at == null) return tr('Resets at midnight.');
    final secs = at.difference((now ?? DateTime.now()).toUtc()).inSeconds;
    if (secs <= 0) return tr('Resets at midnight.');
    return tr('Resets at {t} (in {d}).', {'t': clockText(at), 'd': durationWords(Duration(seconds: secs))});
  }
}

    this.purchaseId,
    this.url,
    this.lastPhase = 'starting',
  });

  String get fingerprint => '$userId|$method|$product|${tokens ?? ''}|${wallet ?? ''}|${account ?? ''}';
  DateTime get createdAt => DateTime.fromMillisecondsSinceEpoch(createdMs);

  Map<String, dynamic> toJson() => {
        'key': key, 'uid': userId, 'product': product, 'tokens': tokens, 'method': method, 'wallet': wallet, 'wallet_label': walletLabel,
        'account': account, 'created': createdMs, 'purchase': purchaseId, 'url': url, 'phase': lastPhase,
      };

  static PendingPurchase? fromJson(Object? v) {
    if (v is! Map) return null;
    final key = v['key'], uid = v['uid'], product = v['product'];
    if (key is! String || uid is! String || product is! String) return null;
    return PendingPurchase(
      key: key,
      userId: uid,
      product: product,
      tokens: (v['tokens'] as num?)?.toInt(),
      method: v['method'] == 'wallet' ? 'wallet' : 'checkout',
      wallet: v['wallet'] as String?,
      walletLabel: v['wallet_label'] as String?,
      account: v['account'] as String?,
      createdMs: (v['created'] as num?)?.toInt() ?? 0,
      purchaseId: v['purchase'] as String?,
      url: v['url'] as String?,
      lastPhase: '${v['phase'] ?? 'starting'}',
    );
  }
}

/// "Done. 3M added." after a payment (the amount it added, when the server said).
String addedText(int tokens) => tokens > 0 ? tr('Done. {x} added.', {'x': tokensShort(tokens)}) : tr('Done. Payment received.');

/// Stops a poll loop (screen closed, user tapped something else).
class PollToken {
  bool cancelled = false;
  void cancel() => cancelled = true;
}

/// Wait before poll number [n] (0-based): starts at the server's hint (default 3 s), grows 1.5x, at most 20 s.
Duration pollDelay(int n, {int? hintSeconds}) {
  final base = (hintSeconds == null || hintSeconds <= 0 ? 3 : hintSeconds).clamp(2, 20).toDouble();
  final s = math.min(20.0, base * math.pow(1.5, n));
  return Duration(milliseconds: (s * 1000).round());
}

  List<PendingPurchase> _mine() {
    final uid = api.userId;
    return uid == null ? const [] : _all.where((r) => r.userId == uid).toList();

  /// The attempt for this exact payment: an unsettled one from earlier (same product, method, wallet and number,
  /// under 24 h old) keeps its idempotency key, so retrying after a crash can never charge twice. Else a new key.
  Future<PendingPurchase> beginAttempt(
      {required String product, int? tokens, required String method, String? wallet, String? walletLabel, String? account}) async {
    await _load();
    final uid = api.userId ?? 'anon';
    final probe = PendingPurchase(key: '', userId: uid, product: product, tokens: tokens, method: method, wallet: wallet, account: account, createdMs: 0);
    final now = _now().millisecondsSinceEpoch;
    for (final r in _all) {
      if (r.fingerprint == probe.fingerprint && now - r.createdMs < pendingMaxAge.inMilliseconds) return r;
    }
    final a = PendingPurchase(
      key: 'app-${_uuid.v4()}',
      userId: uid,
      product: product,
      tokens: tokens,
      method: method,
      wallet: wallet,
      walletLabel: walletLabel,
      account: account,
      createdMs: now,
    );
    _all.add(a);
    await _save(); // persisted BEFORE the request goes out
    return a;
  }

  /// Drops [a]'s key so the next attempt for the same thing gets a fresh one (after a confirmed refusal).
  Future<void> forgetAttempt(PendingPurchase a) async {
    _all.removeWhere((r) => r.key == a.key);
    await _save();
  }

  /// Starts (or, with the same key, safely re-asks about) the payment. A second tap while one is in flight gets the
  /// same answer: one request per key at a time.
  Future<PayUpdate> submit(PendingPurchase a) {
    final running = _inFlight[a.key];
    if (running != null) return running;
    // (block body: returning the removed future from whenComplete would make it wait on itself)
    final f = _submit(a).whenComplete(() {
      _inFlight.remove(a.key);
    });
    _inFlight[a.key] = f;
    return f;
  }

  Future<PayUpdate> _submit(PendingPurchase a) async {
    final body = <String, dynamic>{
      'plan': a.product,
      if (a.tokens != null) 'tokens': a.tokens,
      if (a.method == 'wallet') ...{'method': 'wallet', 'wallet': a.wallet, 'account': a.account},
    };
    final dev = await api.deviceId();
    if (dev != null && dev.length >= 6) body['device_id'] = dev;
    PayUpdate u;
    try {
      final r = await api.postCheckout(body, idempotencyKey: a.key);
      u = PayUpdate.fromHttp(r.status, r.body, purchaseId: a.purchaseId, replay: r.headers['x-idempotent-replay'] == 'true');
    } on BillingNetworkError catch (e) {
      u = PayUpdate(PayPhase.offline, purchaseId: a.purchaseId, reason: e.kind, message: PayText.messageFor(PayPhase.offline, e.kind));
    } catch (e) {
      debugPrint('billing: checkout: $e');
      u = PayUpdate(PayPhase.offline, purchaseId: a.purchaseId, reason: 'offline', message: PayText.messageFor(PayPhase.offline, 'offline'));
    }
    await _record(a, u);
    return u;
  }

  /// One status check (GET billing-checkout?purchase=). The server re-verifies with the provider at most every 10 s.
  Future<PayUpdate> check(String purchaseId) async {
    PayUpdate u;
    try {
      final r = await api.getStatus(purchaseId);
      u = PayUpdate.fromHttp(r.status, r.body, purchaseId: purchaseId);
      if (r.status == 404) u = PayUpdate(PayPhase.error, purchaseId: purchaseId, reason: 'not_found', message: PayText.messageFor(PayPhase.error, 'not_found'), httpStatus: 404);
    } on BillingNetworkError catch (e) {
      u = PayUpdate(PayPhase.offline, purchaseId: purchaseId, reason: e.kind, message: PayText.messageFor(PayPhase.offline, e.kind));
    } catch (e) {
      u = PayUpdate(PayPhase.offline, purchaseId: purchaseId, reason: 'offline', message: PayText.messageFor(PayPhase.offline, 'offline'));
    }
    final a = _byPurchase(purchaseId);
    if (a != null) await _record(a, u);
    return u;
  }

  /// Polls until the payment settles, with backoff. Stops after [maxWait] / [maxPolls] and answers
  /// [PayPhase.stillWaiting] (the screen then offers "Check now"; foregrounding the app resumes polling).
  /// Being offline for a while does not end the loop early; it reports [PayPhase.offline] and keeps trying.
  Future<PayUpdate> poll(String purchaseId,
      {void Function(PayUpdate)? onUpdate, PollToken? token, Duration maxWait = defaultPollWait, int maxPolls = defaultMaxPolls, int? hintSeconds}) async {
    final start = _now();
    PayUpdate last = PayUpdate(PayPhase.processing, purchaseId: purchaseId, message: PayText.messageFor(PayPhase.processing, null));
    for (var n = 0; n < maxPolls; n++) {
      if (token?.cancelled == true) return last;
      await _sleep(pollDelay(n, hintSeconds: hintSeconds ?? last.pollAfterSeconds));
      if (token?.cancelled == true) return last;
      final u = await check(purchaseId);
      // A failed status read is not a failed payment: keep the last real phase, but say we are offline.
      last = u;
      onUpdate?.call(u);
      if (u.settled) return u;
      if (u.phase == PayPhase.error && u.reason == 'not_found') return u;
      if (u.phase == PayPhase.error && u.reason == 'auth_required') return u;
      if (_now().difference(start) >= maxWait) break;
    }
    final w = PayUpdate(PayPhase.stillWaiting,
        purchaseId: purchaseId, reason: last.reason, message: PayText.messageFor(PayPhase.stillWaiting, null), httpStatus: last.httpStatus);
    onUpdate?.call(w);
    return w;
  }

  /// The billing-return URL the payment page was sent to: let the server see it (binds the payment ref).
  Future<void> touchReturn(Uri u) => api.touchReturn(u);

  String get functionsHost => api.functionsHost;

  /// A submac://pay link (the payment page was opened outside the app, or the in-app page missed the redirect).
  /// Never trusted: it only makes the app check that purchase now.
  Future<void> handlePayLink(Uri u) async {
    final d = parsePayDeepLink(u);
    if (d == null) return;
    resumeTick.value++;
    final id = d.purchaseId;
    if (id == null) return;
    final known = _byPurchase(id);
    final r = await check(id);
    if (known == null && r.phase == PayPhase.success) {
      onToast?.call(addedText(r.tokensAdded));
    }
  }

  /// Checks every unsettled payment of this user (after a restart, on foreground). Settled ones are dropped and,
  /// when the app had lost track of them, announced in a snackbar.
  Future<void> recoverPending({bool announce = true}) async {
    if (_recovering) return;
    _recovering = true;
    try {
      await _load();
      final now = _now().millisecondsSinceEpoch;
      for (final r in _mine()) {
        if (_inFlight.containsKey(r.key)) continue;
        final old = now - r.createdMs > pendingMaxAge.inMilliseconds;
        final pid = r.purchaseId;
        if (pid == null) {
          // Never got an answer before the app died. Only a tap re-sends it (re-sending on its own could start a
          // wallet prompt the user no longer expects). Forgotten after 24 h; the server expires it too.
          if (old) await forgetAttempt(r);
          continue;
        }
        final u = await check(pid);
        if (u.phase == PayPhase.success) {
          if (announce) onToast?.call(addedText(u.tokensAdded));
        } else if (u.phase == PayPhase.failed) {
          if (announce && r.lastPhase != 'failed') onToast?.call('A payment you started didn\'t go through. No money was taken.');
        } else if (old || (u.phase == PayPhase.error && u.reason == 'not_found')) {
          await forgetAttempt(r);
        }
      }
    } finally {
      _recovering = false;
    }
  }

  /// Unsettled payments to show in the wallet ("Payment being confirmed · Check now").
  List<PendingPurchase> get unsettled => _mine();

  PendingPurchase? _byPurchase(String id) {
    for (final r in _all) {
      if (r.purchaseId == id) return r;
    }
    return null;
  }

  Future<void> _record(PendingPurchase a, PayUpdate u) async {
    if (u.purchaseId != null) a.purchaseId = u.purchaseId;
    if (u.url != null) a.url = u.url;
    a.lastPhase = u.phase.name;
    if (u.settled || PayUpdate.freshKeyReasons.contains(u.reason) || (u.phase == PayPhase.error && a.purchaseId == null && u.httpStatus > 0 && u.httpStatus < 500)) {
      // Done (or refused before any purchase existed): the next attempt gets a fresh key.
      _all.removeWhere((r) => r.key == a.key);
    }
    await _save();
    if (u.phase == PayPhase.success) unawaited(refreshUsage());
  }

  Future<void> _load() async {
    if (_loaded) return;
    try {
      final p = await _p();
      final raw = p.getString(_pendingKey);
      if (raw != null) {
        final l = jsonDecode(raw);
        if (l is List) {
          for (final x in l) {
            final r = PendingPurchase.fromJson(x);
            if (r != null && !_all.any((e) => e.key == r.key)) _all.add(r);
          }
        }
      }
    } catch (e) {
      debugPrint('billing: load pending: $e');
    }
    _loaded = true;
    pending.value = List.unmodifiable(_mine());
  }

  bool _loaded = false;

  Future<void> _save() async {
    pending.value = List.unmodifiable(_mine());
    try {
      final p = await _p();
      await p.setString(_pendingKey, jsonEncode([for (final r in _all) r.toJson()]));
    } catch (e) {
      debugPrint('billing: save pending: $e');
    }
  }

  Future<void> _saveSummary(Map<String, dynamic> j) async {
    try {
      final uid = api.userId;
      if (uid == null) return;
      await (await _p()).setString(_summaryKey, jsonEncode({'uid': uid, 'at': _now().toIso8601String(), 'summary': j}));
    } catch (_) {}
  }

  Future<void> _loadSummary() async {
    try {
      final raw = (await _p()).getString(_summaryKey);
      if (raw == null) return;
      final j = jsonDecode(raw);
      if (j is! Map || j['uid'] != api.userId || j['summary'] is! Map) return;
      usage.value = UsageSummary.fromJson((j['summary'] as Map).map((k, v) => MapEntry('$k', v)),
          at: DateTime.tryParse('${j['at']}') ?? _now());
    } catch (_) {}
  }

  /// Signed out: forget the cached balance (pending purchases stay; they belong to that account and are only
  /// checked while it is signed in).
  void signedOut() {
    usage.value = null;
    balance.value = null;
    enabled.value = false;
    pending.value = const [];
  }
}

import 'package:flutter/foundation.dart';

/// What the in-app payment page does with a navigation (hosted card checkout on pay.sifalo.net).
enum CheckoutNavKind {
  /// Load it in the payment page.
  allow,

  /// The payment page is done (billing-return URL or submac://pay): stop loading, close, and check the purchase.
  finish,

  /// Not a payment page: open it outside the app (browser, phone, mail).
  external,

  /// Never load it, never open it (javascript:, file:, plain http to a payment host...).
  block,
}

@immutable
class CheckoutNavDecision {
  final CheckoutNavKind kind;

  /// From the return URL (purchase= or order_id=). Informational only: the app always checks ITS OWN purchase id.
  final String? purchaseId;

  /// What the redirect claimed (paid | pending | failed | error). Never trusted: the app polls billing-checkout.
  final String? status;
  final String? reason;

  /// For [CheckoutNavKind.finish] on the billing-return URL: the server must still see this request (it binds the
  /// payment ref Sifalo put in the URL), so the app sends it itself in the background.
  final Uri? serverReturnUrl;
  const CheckoutNavDecision(this.kind, {this.purchaseId, this.status, this.reason, this.serverReturnUrl});

  @override
  String toString() => 'CheckoutNavDecision($kind, purchase: $purchaseId, status: $status)';
}

/// Hosts the payment page may load. Owner rule (2026-10-21): only Sifalo's pay hosts and our own functions host.
const kCheckoutPayHosts = {'pay.sifalo.net', 'pay.sifalo.com'};

/// Extra hosts allowed inside the payment page (e.g. a card network's 3-D Secure page, if staging shows the hosted
/// checkout redirects the whole page there). Empty on purpose until seen on staging.
const Set<String> kCheckoutExtraHosts = {};

/// Decides what happens when the payment page wants to go to [u].
/// [functionsHost] is the Supabase project host (xyz.supabase.co). [mainFrame] = false for iframes: those may load any
/// https page (card 3-D Secure challenges live in iframes) but never leave the app or finish the flow.
CheckoutNavDecision decideCheckoutNav(Uri u, {required String functionsHost, bool mainFrame = true}) {
  final scheme = u.scheme.toLowerCase();
  final host = u.host.toLowerCase();
  if (scheme == 'submac') {
    if (host == 'pay') return _finish(u);
    return const CheckoutNavDecision(CheckoutNavKind.block);
  }
  if (scheme == 'about') return const CheckoutNavDecision(CheckoutNavKind.allow); // about:blank / about:srcdoc frames
  if (scheme == 'https') {
    if (host == functionsHost.toLowerCase() && isBillingReturnPath(u.path)) {
      return mainFrame ? _finish(u, server: u) : const CheckoutNavDecision(CheckoutNavKind.block);
    }
    if (kCheckoutPayHosts.contains(host) || kCheckoutExtraHosts.contains(host) || host == functionsHost.toLowerCase()) {
      return const CheckoutNavDecision(CheckoutNavKind.allow);
    }
    return CheckoutNavDecision(mainFrame ? CheckoutNavKind.external : CheckoutNavKind.allow);
  }
  if (scheme == 'http') {
    // Never load a payment page without TLS; other plain links leave the app.
    if (kCheckoutPayHosts.contains(host) || host == functionsHost.toLowerCase()) return const CheckoutNavDecision(CheckoutNavKind.block);
    return CheckoutNavDecision(mainFrame ? CheckoutNavKind.external : CheckoutNavKind.block);
  }
  if (!mainFrame) {
    // Frames may use data:/blob: for their own content.
    if (scheme == 'data' || scheme == 'blob') return const CheckoutNavDecision(CheckoutNavKind.allow);
    return const CheckoutNavDecision(CheckoutNavKind.block);
  }
  if (const {'tel', 'mailto', 'sms', 'whatsapp', 'intent', 'market'}.contains(scheme)) {
    return const CheckoutNavDecision(CheckoutNavKind.external);
  }
  return const CheckoutNavDecision(CheckoutNavKind.block); // javascript:, file:, data: top-level, unknown schemes
}

/// `/functions/v1/billing-return` (optionally `/functions/v1/billing-return/<provider>`).
bool isBillingReturnPath(String path) {
  final p = path.toLowerCase();
  return p == '/functions/v1/billing-return' || p.startsWith('/functions/v1/billing-return/') || p == '/functions/v1/billing-return/';
}

CheckoutNavDecision _finish(Uri u, {Uri? server}) {
  final q = u.queryParameters;
  final id = q['purchase'] ?? q['order_id'];
  return CheckoutNavDecision(
    CheckoutNavKind.finish,
    purchaseId: id != null && _uuid.hasMatch(id) ? id.toLowerCase() : null,
    status: q['status'],
    reason: q['reason'],
    serverReturnUrl: server,
  );
}

final _uuid = RegExp(r'^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$');

/// A submac://pay deep link (fallback when the page was opened outside the app): the purchase it is about.
CheckoutNavDecision? parsePayDeepLink(Uri u) {
  if (u.scheme.toLowerCase() != 'submac' || u.host.toLowerCase() != 'pay') return null;
  return _finish(u);
}

import 'dart:convert';

import 'package:flutter/foundation.dart';

import 'billing_copy.dart';

/// Where one payment attempt stands, as the payment screen shows it.
enum PayPhase {
  idle,

  /// Asking the server to start the payment.
  starting,

  /// Wallet push sent: "approve on your phone".
  approve,

  /// Hosted card page is open (in the in-app payment page).
  checkout,

  /// The server / provider is confirming (slow or briefly down). Money may already be on its way.
  processing,

  /// Polling stopped after its limit without an answer: "I've approved, check now".
  stillWaiting,

  /// Paid and added to the user's plan.
  success,

  /// Definitely not paid (declined, insufficient funds, expired, reversed, refused).
  failed,

  /// Could not reach SubMac at all. Safe to retry with the same key.
  offline,

  /// Something to fix before paying (bad number, rate limit, payments off...). Nothing was charged.
  error,
}

/// One answer from billing-checkout (POST start or GET ?purchase= poll), or a network failure, in app terms.
@immutable
class PayUpdate {
  final PayPhase phase;
  final String? purchaseId;
  final String? reason;
  final String message;
  final String? url;
  /// Tokens this payment added (0 until paid).
  final int tokensAdded;
  final int? pollAfterSeconds;
  final int httpStatus;

  /// The server's own answer was a replay of an earlier request with the same idempotency key.
  final bool replay;
  const PayUpdate(this.phase,
      {this.purchaseId,
      this.reason,
      this.message = '',
      this.url,
      this.tokensAdded = 0,
      this.pollAfterSeconds,
      this.httpStatus = 0,
      this.replay = false});

  bool get settled => phase == PayPhase.success || phase == PayPhase.failed;
  bool get pending => phase == PayPhase.approve || phase == PayPhase.processing || phase == PayPhase.checkout || phase == PayPhase.stillWaiting;

  /// Retrying with the SAME idempotency key is right whenever the outcome is unknown or the request never got an answer:
  /// the server returns the same purchase and never charges twice. After a confirmed refusal (declined, not enough
  /// money, expired...) the same key would only replay that refusal, so a retry needs a fresh attempt.
  bool get retryKeepsKey => !(phase == PayPhase.failed || freshKeyReasons.contains(reason));

  static const freshKeyReasons = {'idempotency_key_reused', 'payment_mismatch', 'bad_idempotency_key'};

  PayUpdate copyWith({PayPhase? phase, String? message, String? purchaseId, String? reason}) => PayUpdate(phase ?? this.phase,
      purchaseId: purchaseId ?? this.purchaseId,
      reason: reason ?? this.reason,
      message: message ?? this.message,
      url: url,
      tokensAdded: tokensAdded,
      pollAfterSeconds: pollAfterSeconds,
      httpStatus: httpStatus,
      replay: replay);

  @override
  String toString() => 'PayUpdate($phase, purchase: $purchaseId, reason: $reason, http: $httpStatus)';

  /// Parses any billing-checkout answer. Never throws: an unreadable body becomes [PayPhase.processing] when the
  /// server may have taken the payment (5xx / 202), else [PayPhase.error].
  static PayUpdate fromHttp(int status, String body, {String? purchaseId, bool replay = false}) {
    Map<String, dynamic> j = const {};
    try {
      final d = jsonDecode(body);
      if (d is Map) j = d.map((k, v) => MapEntry('$k', v));
    } catch (_) {}
    final pid = _s(j['purchase_id']) ?? purchaseId;
    final state = _s(j['state']);
    final reason = _s(j['reason']) ?? _s(j['error']) ?? _s(j['result']);
    final serverMsg = _s(j['message']) ?? '';
    final added = j['tokens_added'] is num ? (j['tokens_added'] as num).toInt() : 0;
    final poll = j['poll_after_seconds'] is num ? (j['poll_after_seconds'] as num).toInt() : null;
    final rp = replay || j['replay'] == true;
    PayUpdate mk(PayPhase p, {String? r}) => PayUpdate(p,
        purchaseId: pid,
        reason: r ?? reason,
        message: PayText.messageFor(p, r ?? reason, serverMsg),
        url: _s(j['url']),
        tokensAdded: added,
        pollAfterSeconds: poll,
        httpStatus: status,
        replay: rp);
    switch (state) {
      case 'paid':
        return mk(PayPhase.success);
      case 'waiting_approval':
        return mk(PayPhase.approve);
      case 'checkout_opened':
        return _s(j['url']) != null ? mk(PayPhase.checkout) : mk(PayPhase.processing);
      case 'processing':
        return mk(PayPhase.processing);
      case 'failed':
        return mk(PayPhase.failed, r: reason ?? 'declined');
      case 'expired':
        return mk(PayPhase.failed, r: 'expired');
      case 'reversed':
        return mk(PayPhase.failed, r: 'reversed');
      case 'error':
        return mk(PayPhase.error, r: reason ?? 'payment_mismatch');
    }
    // No state: a refusal before any purchase existed, or a body we can't read.
    if (status == 202) return mk(PayPhase.processing);
    if (status >= 200 && status < 300 && j['ok'] == true && j['status'] == 'paid') return mk(PayPhase.success);
    if (status == 0) return mk(PayPhase.offline, r: reason ?? 'offline');
    if (status >= 500 && pid != null) return mk(PayPhase.processing, r: reason ?? 'provider_unavailable');
    if (status == 401) return mk(PayPhase.error, r: 'auth_required');
    return mk(PayPhase.error, r: reason ?? (status >= 500 ? 'service_unavailable' : 'error'));
  }

  static String? _s(Object? v) => v == null || '$v'.isEmpty ? null : '$v';
}

/// How the in-app payment page was left.
enum CheckoutExitKind { returned, cancelled, failedToLoad }

@immutable
class CheckoutExit {
  final CheckoutExitKind kind;

  /// When the page itself already checked the purchase (close button: "Cancel payment?" then a status check).
  final PayUpdate? checked;
  const CheckoutExit(this.kind, {this.checked});
}

/// Opens the hosted payment page (in-app WebView) and resolves when it closes.
typedef OpenCheckout = Future<CheckoutExit> Function(String url, String purchaseId);

/// Drives one payment from tap to result: start, approve-on-phone countdown, polling with backoff, retry with the same
/// idempotency key, in-app card page, foreground resume. UI-free so it can be unit tested.
class PaymentFlow extends ChangeNotifier {
  PaymentFlow(this.billing, this.request, {this.openCheckout, this.approveWindow = const Duration(seconds: 120), DateTime Function()? now})
      : _now = now ?? DateTime.now {
    billing.resumeTick.addListener(_onResume);
  }

  final Billing billing;
  final PaymentRequest request;
  final OpenCheckout? openCheckout;
  final Duration approveWindow;
  final DateTime Function() _now;

  PayUpdate state = const PayUpdate(PayPhase.idle);

  /// A request or poll is running: buttons that would start another one are disabled.
  bool busy = false;

  /// The last status read failed for lack of connection (the payment itself may be fine).
  bool offlineNow = false;

  /// When the "approve on your phone" window ends (wallet payments).
  DateTime? approveDeadline;

  PendingPurchase? _attempt;
  PollToken? _poll;
  bool _disposed = false;
  bool _pageOpen = false;

  String? get purchaseId => state.purchaseId ?? _attempt?.purchaseId;
  String? get attemptKey => _attempt?.key;
  bool get polling => _poll != null && !_poll!.cancelled;

  Duration get approveLeft {
    final d = approveDeadline;
    if (d == null) return Duration.zero;
    final left = d.difference(_now());
    return left.isNegative ? Duration.zero : left;
  }

  void _set(PayUpdate u) {
    if (_disposed) return;
    if (u.phase == PayPhase.offline && state.purchaseId != null && state.pending) {
      // Lost the connection while checking: keep showing where the payment stood.
      offlineNow = true;
    } else {
      offlineNow = u.phase == PayPhase.offline;
      state = u;
    }
    if (u.phase == PayPhase.approve && approveDeadline == null) approveDeadline = _now().add(approveWindow);
    notifyListeners();
  }

  /// Starts the payment. A second call while one is running does nothing (double-tap protection).
  Future<void> start() async {
    if (busy || _disposed) return;
    busy = true;
    _set(PayUpdate(PayPhase.starting, message: PayText.messageFor(PayPhase.starting, null)));
    try {
      _attempt ??= await billing.beginAttempt(
        product: request.planId,
        tokens: request.tokens,
        method: request.method,
        wallet: request.gateway,
        walletLabel: request.walletLabel,
        account: request.account,
      );
      final known = _attempt!.purchaseId;
      // Already have a purchase for this attempt (started before a crash / from the wallet): ask, don't re-send.
      final u = known != null ? await billing.check(known) : await billing.submit(_attempt!);
      await _handle(u);
    } finally {
      busy = false;
      if (!_disposed) notifyListeners();
    }
  }

  /// The single retry button. Same idempotency key while the outcome is unknown (offline, timeout, provider down,
  /// still waiting): the server returns the same purchase and never charges twice. After a confirmed refusal
  /// (declined, not enough money, expired) a fresh attempt is started, since the old key would only replay the refusal.
  Future<void> retry() async {
    if (busy || _disposed) return;
    final a = _attempt;
    if (a != null && !state.retryKeepsKey) {
      await billing.forgetAttempt(a);
      _attempt = null;
      approveDeadline = null;
    } else if (a != null && a.purchaseId != null && state.pending) {
      // Known purchase still open: checking is the retry (re-sending would only replay it).
      if (request.method == 'checkout' && (a.url ?? state.url) != null && openCheckout != null) {
        busy = true;
        try {
          await _openPage(a.url ?? state.url!, a.purchaseId!);
        } finally {
          busy = false;
          if (!_disposed) notifyListeners();
        }
        return;
      }
      return checkNow();
    }
    return start();
  }

  /// "I've approved, check now" / "I've paid, check now".
  Future<void> checkNow() async {
    final pid = purchaseId;
    if (pid == null || busy || _disposed) return;
    _poll?.cancel();
    busy = true;
    notifyListeners();
    try {
      final u = await billing.check(pid);
      _set(u);
      if (!u.settled && (u.pending || u.phase == PayPhase.offline)) unawaited(_startPolling(pid));
    } finally {
      busy = false;
      if (!_disposed) notifyListeners();
    }
  }

  Future<void> _handle(PayUpdate u) async {
    _set(u);
    final pid = u.purchaseId;
    if (u.phase == PayPhase.checkout && u.url != null && pid != null) {
      await _openPage(u.url!, pid);
      return;
    }
    if ((u.phase == PayPhase.approve || u.phase == PayPhase.processing) && pid != null) {
      unawaited(_startPolling(pid, hint: u.pollAfterSeconds));
    }
  }

  Future<void> _openPage(String url, String pid) async {
    final open = openCheckout;
    if (open == null || _pageOpen) return;
    _pageOpen = true;
    CheckoutExit exit;
    try {
      exit = await open(url, pid);
    } catch (_) {
      exit = const CheckoutExit(CheckoutExitKind.failedToLoad);
    } finally {
      _pageOpen = false;
    }
    if (_disposed) return;
    // Never trust the redirect: ask the server what happened to OUR purchase.
    final u = exit.checked ?? await billing.check(pid);
    if (u.settled) {
      _set(u);
      return;
    }
    if (exit.kind == CheckoutExitKind.cancelled) {
      _set(PayUpdate(PayPhase.stillWaiting,
          purchaseId: pid,
          url: url,
          message: 'Payment not finished. If you already paid, tap Check now. Otherwise you can try again.'));
      return;
    }
    _set(u.phase == PayPhase.offline ? u : PayUpdate(PayPhase.processing, purchaseId: pid, url: url, message: PayText.messageFor(PayPhase.processing, null)));
    unawaited(_startPolling(pid, hint: 2));
  }

  Future<void> _startPolling(String pid, {int? hint}) async {
    _poll?.cancel();
    final t = PollToken();
    _poll = t;
    if (!_disposed) notifyListeners();
    await billing.poll(pid, token: t, hintSeconds: hint, onUpdate: (u) {
      if (!t.cancelled) _set(u);
    });
    if (identical(_poll, t)) _poll = null;
    if (!_disposed) notifyListeners();
  }

  void _onResume() {
    // Back from the wallet app / USSD prompt: look again right away, then keep polling.
    final pid = purchaseId;
    if (pid == null || _disposed || busy || _pageOpen) return;
    if (state.pending || state.phase == PayPhase.offline || offlineNow) unawaited(checkNow());
  }

  @override
  void dispose() {
    _disposed = true;
    _poll?.cancel();
    billing.resumeTick.removeListener(_onResume);
    super.dispose();
  }
}

/// The result of checking a wallet number as typed.
@immutable
class WalletNumberCheck {
  /// 2526XXXXXXXX (12 digits) when the number is a valid Somali mobile, else null.
  final String? normalized;

  /// Why it can't be used (shown under the field). null = OK.
  final String? error;

  /// A gentle hint, never blocking: "This looks like an eDahab number".
  final String? hint;

  /// The wallet the number most likely belongs to (null when unknown or already the chosen one).
  final MobileWallet? suggested;
  const WalletNumberCheck({this.normalized, this.error, this.hint, this.suggested});
  bool get ok => normalized != null && error == null;
}

/// Turns what people type (+252 61 234 5678, 0612345678, 612345678, 00252...) into 2526XXXXXXXX.
String? normalizeSomaliMobile(String input) {
  var d = input.replaceAll(RegExp(r'[^0-9]'), '');
  if (d.startsWith('00')) d = d.substring(2);
  if (d.startsWith('252')) {
    d = d.substring(3);
    if (d.startsWith('0')) d = d.substring(1); // +252 0612...
  } else if (d.startsWith('0')) {
    d = d.substring(1);
  }
  if (d.length != 9) return null;
  // Somali mobiles start with 6, 7 or 9 (Golis 90).
  if (!RegExp(r'^(6[1-9]|7[0-9]|9[0-9])[0-9]{7}$').hasMatch(d)) return null;
  return '252$d';
}

/// The wallet a number's prefix points to (null = no clear match).
MobileWallet? walletForNumber(String normalized) {
  if (normalized.length != 12) return null;
  final p = normalized.substring(3, 5);
  for (final w in MobileWallet.all) {
    if (w.prefixes.contains(p)) return w;
  }
  return null;
}

/// Checks [input] for [wallet]. A format problem blocks; an operator mismatch only hints (prefix lists change, numbers port).
WalletNumberCheck checkWalletNumber(String input, MobileWallet wallet) {
  final raw = input.trim();
  if (raw.isEmpty) return WalletNumberCheck(error: tr('Enter your wallet number.'));
  final digits = raw.replaceAll(RegExp(r'[^0-9]'), '');
  if (RegExp(r'[A-Za-z]').hasMatch(raw)) return WalletNumberCheck(error: tr('Use digits only, like 61 234 5678.'));
  final n = normalizeSomaliMobile(raw);
  if (n == null) {
    final national = digits.startsWith('252') ? digits.substring(3) : digits.startsWith('0') ? digits.substring(1) : digits;
    if (national.length < 9) return WalletNumberCheck(error: tr('This number is too short.'));
    if (national.length > 9) return WalletNumberCheck(error: tr('This number is too long.'));
    return WalletNumberCheck(error: tr("This doesn't look like a Somali mobile number."));
  }
  final likely = walletForNumber(n);
  // Same gateway (EVC / ZAAD / SAHAL are all Waafi) or a wallet that works on any number: fine.
  if (wallet.prefixes.isEmpty || likely == null || likely.gateway == wallet.gateway) {
    return WalletNumberCheck(normalized: n);
  }
  return WalletNumberCheck(
    normalized: n,
    hint: tr('This looks like a {x} number. Make sure it has {y}.', {'x': '${likely.label} (${likely.operator})', 'y': wallet.label}),
    suggested: likely,
  );
}

/// "252 61 234 5678" for display.
String formatSomaliMobile(String normalized) {
  if (normalized.length != 12) return normalized;
  return '+252 ${normalized.substring(3, 5)} ${normalized.substring(5, 8)} ${normalized.substring(8)}';
}

/// "****5678", for logs and lists.
String maskNumber(String n) => n.length <= 4 ? n : '****${n.substring(n.length - 4)}';

class _BurstPainter extends CustomPainter {
  _BurstPainter(this.t, this.colors);
  final double t;
  final List<Color> colors;
  @override
  void paint(Canvas canvas, Size size) {
    if (t <= 0 || t >= 1) return;
    final c = size.center(Offset.zero);
    final ease = Curves.easeOutCubic.transform(t);
    for (var i = 0; i < 18; i++) {
      final a = i * (math.pi * 2 / 18) + (i.isEven ? 0.12 : -0.08);
      final r = size.width * 0.12 + size.width * 0.36 * ease * (0.75 + (i % 3) * 0.12);
      final o = c + Offset(math.cos(a) * r, math.sin(a) * r + 18 * t * t);
      final paint = Paint()..color = colors[i % colors.length].withValues(alpha: (1 - t).clamp(0.0, 1.0));
      canvas.drawCircle(o, 3.2 - 1.6 * t, paint);
    }
  }

  @override
  bool shouldRepaint(_BurstPainter old) => old.t != t;
}

/// What the payment page needs from whatever web view shows it (the real InAppWebView, or a fake in tests).
abstract class CheckoutWebController {
  Future<void> reload();
  Future<bool> canGoBack();
  Future<void> goBack();
  Future<void> stopLoading();
}

/// Callbacks the web view reports into the payment page.
class CheckoutWebHooks {
  CheckoutWebHooks({
    required this.url,
    required this.onCreated,
    required this.onNavigate,
    required this.onLoadStart,
    required this.onProgress,
    required this.onMainFrameError,
    required this.onLoadStop,
  });
  final String url;
  final void Function(CheckoutWebController c) onCreated;

  /// Every navigation (links, redirects, new windows). Returns true to let it load in the page.
  final Future<bool> Function(Uri uri, {bool mainFrame}) onNavigate;

  /// A page started loading (some redirects only show up here): returns true when the page is done.
  final bool Function(Uri uri) onLoadStart;
  final void Function(int progress) onProgress;
  final void Function(String description) onMainFrameError;
  final void Function() onLoadStop;
}

class _CheckoutWebViewScreenState extends State<CheckoutWebViewScreen> {
  CheckoutWebController? _web;
  int _progress = 0;
  String? _error;
  bool _done = false;
  bool _checking = false;

  Future<bool> _navigate(Uri uri, {bool mainFrame = true}) async {
    if (_done) return false;
    final d = decideCheckoutNav(uri, functionsHost: widget.functionsHost, mainFrame: mainFrame);
    switch (d.kind) {
      case CheckoutNavKind.allow:
        return true;
      case CheckoutNavKind.finish:
        _finish(d);
        return false;
      case CheckoutNavKind.external:
        unawaited(_external(uri));
        return false;
      case CheckoutNavKind.block:
        return false;
    }
  }

  bool _loadStart(Uri uri) {
    if (_done) return true;
    if (_error != null) setState(() => _error = null);
    final d = decideCheckoutNav(uri, functionsHost: widget.functionsHost);
    if (d.kind == CheckoutNavKind.finish) {
      _finish(d);
      return true;
    }
    if (d.kind == CheckoutNavKind.external || d.kind == CheckoutNavKind.block) {
      // A redirect we would not have allowed slipped past the override hook: stop it here.
      unawaited(_web?.stopLoading());
      if (d.kind == CheckoutNavKind.external) unawaited(_external(uri));
    }
    return false;
  }

  void _finish(CheckoutNavDecision d) {
    if (_done) return;
    _done = true;
    unawaited(_web?.stopLoading());
    final ret = d.serverReturnUrl;
    if (ret != null && widget.onReturnUrl != null) unawaited(widget.onReturnUrl!(ret));
    if (mounted) Navigator.of(context).pop(const CheckoutExit(CheckoutExitKind.returned));
  }

  Future<void> _external(Uri uri) async {
    try {
      if (widget.openExternal != null) {
        await widget.openExternal!(uri);
      } else {
        await launchUrl(uri, mode: LaunchMode.externalApplication);
      }
    } catch (_) {}
  }

  Future<void> _close() async {
    if (_done || _checking) return;
    final ok = await showDialog<bool>(
      context: context,
      builder: (d) => AlertDialog(
        title: Text(tr('Cancel payment?')),
        content: Text(tr("If you already paid, it still counts. We'll check before you leave.")),
        actions: [
          TextButton(key: const ValueKey('checkout-keep'), onPressed: () => Navigator.pop(d, false), child: Text(tr('Keep paying'))),
          TextButton(key: const ValueKey('checkout-cancel'), onPressed: () => Navigator.pop(d, true), child: Text(tr('Cancel payment'))),
        ],
      ),
    );
    if (ok != true || !mounted || _done) return;
    setState(() => _checking = true);
    PayUpdate? u;
    try {
      u = await widget.onCheck().timeout(const Duration(seconds: 25));
    } catch (_) {
      u = null;
    }
    if (!mounted || _done) return;
    _done = true;
    Navigator.of(context).pop(CheckoutExit(CheckoutExitKind.cancelled, checked: u));
  }

  Future<void> _checkNow() async {
    if (_checking || _done) return;
    setState(() => _checking = true);
    PayUpdate? u;
    try {
      u = await widget.onCheck().timeout(const Duration(seconds: 25));
    } catch (_) {}
    if (!mounted || _done) return;
    setState(() => _checking = false);
    if (u != null && u.settled) {
      _done = true;
      Navigator.of(context).pop(CheckoutExit(CheckoutExitKind.returned, checked: u));
      return;
    }
    ScaffoldMessenger.maybeOf(context)?.showSnackBar(SnackBar(
        content: Text(tr(u == null || u.phase == PayPhase.offline
            ? "You're offline. Check your connection and try again."
            : "We haven't received this payment yet. If you just paid, give it a minute."))));
  }

  Future<void> _back() async {
    final w = _web;
    if (_error == null && w != null && await w.canGoBack()) {
      await w.goBack();
      return;
    }
    await _close();
  }

  void _retry() {
    setState(() {
      _error = null;
      _progress = 0;
    });
    unawaited(_web?.reload());
  }

  @override
  Widget build(BuildContext context) {

    final hooks = CheckoutWebHooks(
      url: widget.url,
      onCreated: (c) => _web = c,
      onNavigate: _navigate,
      onLoadStart: _loadStart,
      onProgress: (v) {
        if (mounted) setState(() => _progress = v);
      },
      onMainFrameError: (desc) {
        if (mounted && !_done) setState(() => _error = desc);
      },
      onLoadStop: () {
        if (mounted) setState(() => _progress = 100);
      },
    );
    return PopScope(
      canPop: false,
      onPopInvokedWithResult: (didPop, _) {
        if (!didPop) unawaited(_back());
      },
      child: Scaffold(
        appBar: AppBar(
          automaticallyImplyLeading: false,
          leading: IconButton(key: const ValueKey('checkout-close'), tooltip: 'Close', icon: const Icon(Icons.close_rounded), onPressed: _close),
          titleSpacing: 0,
          title: Row(mainAxisSize: MainAxisSize.min, children: [
            Icon(Icons.lock_rounded, size: 18, color: p.text),
            const SizedBox(width: 8),
            Text(tr('Secure payment'), style: TextStyle(color: p.text, fontSize: AppText.lead, fontWeight: FontWeight.w600)),
          ]),
          bottom: PreferredSize(
            preferredSize: const Size.fromHeight(2),
            child: _progress < 100 && _error == null
                ? LinearProgressIndicator(value: _progress <= 0 ? null : _progress / 100, minHeight: 2, backgroundColor: Colors.transparent, color: p.text)
                : const SizedBox(height: 2),
          ),
        ),
        body: Stack(children: [
          Positioned.fill(child: (widget.webBuilder ?? _inAppWebView)(context, hooks)),
          if (_error != null)
            Positioned.fill(
              child: ColoredBox(

                      Text(tr("Couldn't load the payment page"), textAlign: TextAlign.center, style: Brand.serif(size: AppText.display, color: p.text)),

                      Text(tr('Check your connection, then try again. If you already paid, it still counts.'),
                          textAlign: TextAlign.center, style: TextStyle(color: p.muted, fontSize: AppText.body, height: 1.4)),
                      const SizedBox(height: 18),

class _InAppController implements CheckoutWebController {
  _InAppController(this.c);
  final InAppWebViewController c;
  @override
  Future<void> reload() => c.reload();
  @override
  Future<bool> canGoBack() => c.canGoBack();
  @override

Widget _inAppWebView(BuildContext context, CheckoutWebHooks h) {
  return InAppWebView(
    initialUrlRequest: URLRequest(url: WebUri(h.url)),
    initialSettings: InAppWebViewSettings(
      javaScriptEnabled: true,
      useShouldOverrideUrlLoading: true,
      supportMultipleWindows: true,
      javaScriptCanOpenWindowsAutomatically: true,
      allowFileAccess: false,
      allowContentAccess: false,
      allowFileAccessFromFileURLs: false,
      allowUniversalAccessFromFileURLs: false,
      thirdPartyCookiesEnabled: true,
      clearCache: false,
      cacheEnabled: true,
      transparentBackground: false,
      mediaPlaybackRequiresUserGesture: true,
      supportZoom: false,
    ),
    onWebViewCreated: (c) => h.onCreated(_InAppController(c)),
    shouldOverrideUrlLoading: (c, action) async {
      final u = action.request.url;
      if (u == null) return NavigationActionPolicy.ALLOW;
      final ok = await h.onNavigate(Uri.parse(u.toString()), mainFrame: action.isForMainFrame);
      return ok ? NavigationActionPolicy.ALLOW : NavigationActionPolicy.CANCEL;
    },
    onCreateWindow: (c, action) async {
      // target=_blank / window.open: same rules, but loaded in this page (no second window).
      final u = action.request.url;
      if (u == null) return false;
      final uri = Uri.parse(u.toString());
      if (await h.onNavigate(uri)) await c.loadUrl(urlRequest: URLRequest(url: u));
      return false;
    },
    onLoadStart: (c, u) {
      if (u != null) h.onLoadStart(Uri.parse(u.toString()));
    },
    onProgressChanged: (c, v) => h.onProgress(v),
    onLoadStop: (c, u) => h.onLoadStop(),
    onReceivedError: (c, req, err) {
      if (req.isForMainFrame == false) return;
      if (err.type == WebResourceErrorType.CANCELLED || err.type == WebResourceErrorType.UNSUPPORTED_SCHEME) return;
      final u = Uri.tryParse(req.url.toString());
      if (u != null && u.scheme == 'submac') return;
      h.onMainFrameError(err.description);
    },
  );
}

          value: n.toDouble(),
          min: plan.minTokens.toDouble(),
          max: plan.maxTokens.toDouble(),
          divisions: steps,
          onChanged: (v) => setState(() => _tokens = plan.clampTokens(v.round())),
        ),
      Wrap(alignment: WrapAlignment.center, spacing: Space.x2, runSpacing: Space.x2, children: [
        for (final x in plan.presets)
          ChoiceChip(
            key: ValueKey('billing-payg-preset-$x'),
            label: Text(tokensShort(x)),
            selected: n == x,
            onSelected: (_) => setState(() => _tokens = x),
          ),
      ]),
      const SizedBox(height: Space.x5),
      Center(
        child: Text(tr('Valid {x}', {'x': plan.validityText}), style: TextStyle(color: p.muted, fontSize: AppText.meta)),
      ),
      const SizedBox(height: Space.x4),
      FilledButton(
        key: const ValueKey('billing-payg-buy'),
        style: FilledButton.styleFrom(backgroundColor: p.text, foregroundColor: p.bg, minimumSize: const Size(double.infinity, 52)),
        onPressed: () => _buy(plan, tokens: n),

  }

class ApkBuild {
  ApkBuild._();

  /// The id of the built-in skill (English) and its Somali counterpart.
  static const skillId = 'build-android-apk';
  static const skillIdSo = 'so-build-android-apk';

  /// The marker tests look for in every prompt (part of [rule]).
  static const marker = 'Android Studio is only an editor';

  /// One line for the shared house rules (every prompt: main agent, quick replies, Live, bots, teams).
  static const rule = '- Android apps (APK or AAB) can be built remotely on GitHub Actions with no computer: GitHub\'s ubuntu-latest '
      'machines already have the Android SDK, build-tools, platform-tools and Java (ANDROID_HOME is set). $marker and is never '
      'needed; SubMac itself is built this way. Never say a remote APK build is impossible or needs Android Studio or a local setup: '
      'offer it (skill $skillId).';

/**
 * "submac/update": in-app updates. New releases are ONE file in Cloudflare R2; older ones are stored in parts (Supabase
 * kept files under 50 MB). Every address is handed to Android's DownloadManager at once: it keeps downloading after the
 * user leaves the app, shows its own notification and survives the app being closed. A single file is checked
 * (sha256) and renamed in place; parts are joined into one file whose sha256 must match the release. Only then can
 * Install open Android's installer (through [UpdateFiles]).
 * State lives in SharedPreferences "submac_update", so a restart picks up where it was.
 */
object AppUpdater {
    private const val PREFS = "submac_update"
    private const val DIR = "updates"
    private val main = Handler(Looper.getMainLooper())
    private val lock = Any()

    private var channel: MethodChannel? = null
    private var watching = false

    fun attach(c: Context, engine: FlutterEngine) {
        val ctx = c.applicationContext
        val ch = MethodChannel(engine.dartExecutor.binaryMessenger, "submac/update")
        channel = ch
        watchScreen(ctx)
        ch.setMethodCallHandler { call, result ->
            try {
                when (call.method) {
                    "info" -> result.success(info(ctx))
                    "start" -> {
                        @Suppress("UNCHECKED_CAST")
                        val urls = (call.argument<List<Any?>>("urls") ?: emptyList()).map { it.toString() }
                        val code = (call.argument<Any?>("code") as? Number)?.toLong() ?: 0L
                        val sha = call.argument<String>("sha256") ?: ""
                        val size = (call.argument<Any?>("size") as? Number)?.toLong() ?: 0L
                        val title = call.argument<String>("title") ?: "SubMac update"
                        result.success(start(ctx, code, urls, sha, size, title))
                    }
                    "status" -> Thread {
                        val r = try { status(ctx) } catch (e: Exception) { mapOf("state" to "failed", "reason" to (e.message ?: "error")) }
                        main.post { result.success(r) }
                    }.start()
                    "cancel" -> { cancel(ctx); result.success(true) }
                    "install" -> Thread {
                        val r = try { install(ctx) } catch (e: Exception) { "error: " + (e.message ?: "could not open the installer") }
                        main.post { result.success(r) }
                    }.start()
                    "openInstallSettings" -> { openInstallSettings(ctx); result.success(true) }
                    else -> result.notImplemented()
                }
            } catch (e: Exception) {
                result.error("failed", e.message ?: e.toString(), null)
            }
        }
    }

    /**
     * The engine outlives the screen, so "close SubMac and open it again" may not restart Dart. Tell Dart each time
     * the SubMac screen is freshly opened, so a pending update window shows again (owner: every start).
     */
    private fun watchScreen(ctx: Context) {
        if (watching) return
        val app = ctx as? Application ?: return
        watching = true
        app.registerActivityLifecycleCallbacks(object : Application.ActivityLifecycleCallbacks {
            override fun onActivityCreated(a: Activity, saved: Bundle?) {
                if (a is MainActivity && saved == null) main.post { try { channel?.invokeMethod("opened", null) } catch (_: Exception) {} }
            }
            override fun onActivityStarted(a: Activity) {}
            override fun onActivityResumed(a: Activity) {}
            override fun onActivityPaused(a: Activity) {}
            override fun onActivityStopped(a: Activity) {}
            override fun onActivitySaveInstanceState(a: Activity, out: Bundle) {}
            override fun onActivityDestroyed(a: Activity) {}
        })
    }

    private fun prefs(ctx: Context): SharedPreferences = ctx.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
    private fun dir(ctx: Context): File = File(ctx.getExternalFilesDir(null) ?: ctx.filesDir, DIR)
    fun finalFile(ctx: Context, code: Long): File = File(dir(ctx), "submac-update-$code.apk")
    private fun dm(ctx: Context) = ctx.getSystemService(Context.DOWNLOAD_SERVICE) as DownloadManager
    private fun ids(ctx: Context): LongArray =
        (prefs(ctx).getString("ids", "") ?: "").split(',').mapNotNull { it.trim().toLongOrNull() }.toLongArray()

    @Suppress("DEPRECATION")
    private fun info(ctx: Context): Map<String, Any?> {
        val pi = ctx.packageManager.getPackageInfo(ctx.packageName, 0)
        val code = if (Build.VERSION.SDK_INT >= 28) pi.longVersionCode else pi.versionCode.toLong()
        return mapOf(
            "code" to code,
            "name" to (pi.versionName ?: ""),
            "canInstall" to canInstall(ctx),
            "sdk" to Build.VERSION.SDK_INT,
            // Sent with in-app feedback (lib/services/feedback.dart).
            "manufacturer" to Build.MANUFACTURER,
            "model" to Build.MODEL,
            "release" to Build.VERSION.RELEASE,
        )
    }

    private fun canInstall(ctx: Context): Boolean =
        Build.VERSION.SDK_INT < 26 || ctx.packageManager.canRequestPackageInstalls()

    /** Starts (or restarts) the download of every part. Old files and downloads are removed first. */
    private fun start(ctx: Context, code: Long, urls: List<String>, sha: String, size: Long, title: String): List<Long> {
        if (code <= 0 || urls.isEmpty() || !Regex("^[0-9a-f]{64}$").matches(sha)) throw IllegalArgumentException("bad release")
        for (u in urls) if (!u.startsWith("https://")) throw IllegalArgumentException("bad address")
        cancel(ctx)
        val d = dir(ctx)
        d.mkdirs()
        val out = ArrayList<Long>()
        for ((i, u) in urls.withIndex()) {
            val req = DownloadManager.Request(Uri.parse(u))
                .setTitle(if (urls.size > 1) "$title (${i + 1}/${urls.size})" else title)
                .setDescription("Downloading the new version of SubMac")
                .setNotificationVisibility(DownloadManager.Request.VISIBILITY_VISIBLE)
                .setAllowedOverMetered(true)
                .setAllowedOverRoaming(true)
                .setMimeType("application/octet-stream")
                .setDestinationInExternalFilesDir(ctx, DIR, "part-$code-$i.bin")
            out.add(dm(ctx).enqueue(req))
        }
        prefs(ctx).edit()
            .putString("ids", out.joinToString(","))
            .putLong("code", code)
            .putString("sha", sha)
            .putLong("size", size)
            .putInt("parts", urls.size)
            .putBoolean("ready", false)
            .putBoolean("notified", false)
            .apply()
        return out
    }

    private fun partFile(ctx: Context, code: Long, i: Int) = File(dir(ctx), "part-$code-$i.bin")

    /**
     * none | running | paused | failed | bad (joined file did not match) | ready. Joins the parts when all are done.
     * Called from Dart and from [UpdateDone]; never on the main thread.
     */
    fun status(ctx: Context): Map<String, Any?> = synchronized(lock) { statusLocked(ctx) }

    private fun statusLocked(ctx: Context): Map<String, Any?> {
        val p = prefs(ctx)
        val code = p.getLong("code", 0L)
        if (code <= 0L) return mapOf("state" to "none")
        val f = finalFile(ctx, code)
        if (p.getBoolean("ready", false) && f.isFile) {
            return mapOf("state" to "ready", "code" to code, "path" to f.absolutePath, "bytes" to f.length(), "total" to f.length())
        }
        val ids = ids(ctx)
        if (ids.isEmpty()) return mapOf("state" to "none", "code" to code)
        var bytes = 0L
        var total = 0L
        var done = 0
        var failed = false
        var paused = false
        var reason = 0
        val local = HashMap<Long, File>()
        val q = DownloadManager.Query().setFilterById(*ids)
        var seen = 0
        dm(ctx).query(q)?.use { c ->
            val iSt = c.getColumnIndex(DownloadManager.COLUMN_STATUS)
            val iSo = c.getColumnIndex(DownloadManager.COLUMN_BYTES_DOWNLOADED_SO_FAR)
            val iTo = c.getColumnIndex(DownloadManager.COLUMN_TOTAL_SIZE_BYTES)
            val iRe = c.getColumnIndex(DownloadManager.COLUMN_REASON)
            val iId = c.getColumnIndex(DownloadManager.COLUMN_ID)
            val iUri = c.getColumnIndex(DownloadManager.COLUMN_LOCAL_URI)
            while (c.moveToNext()) {
                seen++
                val u = if (iUri >= 0) c.getString(iUri) else null
                if (iId >= 0 && u != null && u.startsWith("file:")) Uri.parse(u).path?.let { local[c.getLong(iId)] = File(it) }
                val st = c.getInt(iSt)
                bytes += maxOf(0L, c.getLong(iSo))
                total += maxOf(0L, c.getLong(iTo))
                when (st) {
                    DownloadManager.STATUS_SUCCESSFUL -> done++
                    DownloadManager.STATUS_FAILED -> { failed = true; reason = c.getInt(iRe) }
                    DownloadManager.STATUS_PAUSED -> { paused = true; reason = c.getInt(iRe) }
                }
            }
        }
        val want = p.getLong("size", 0L)
        if (want > 0) total = want
        // A download the user removed from the system's Downloads list counts as stopped.
        if (seen < ids.size) failed = true
        if (failed) return mapOf("state" to "failed", "code" to code, "bytes" to bytes, "total" to total, "reason" to reason)
        if (done < ids.size) return mapOf("state" to if (paused) "paused" else "running", "code" to code, "bytes" to bytes, "total" to total, "reason" to reason)
        // Every part is in: join them in order and check the whole file.
        val parts = p.getInt("parts", ids.size)
        val sha = p.getString("sha", "") ?: ""
        val tmp = File(dir(ctx), "joining-$code.tmp")
        if (parts == 1) {
            // One file (Cloudflare R2): check it where it landed and rename it; no copy needed.
            val pf = ids.getOrNull(0)?.let { local[it] }?.takeIf { it.isFile } ?: partFile(ctx, code, 0)
            if (!pf.isFile) return mapOf("state" to "failed", "code" to code, "reason" to -1)
            val got1 = sha256(pf)
            if (got1 != sha) {
                pf.delete()
                try { dm(ctx).remove(*ids) } catch (_: Exception) {}
                p.edit().putString("ids", "").putLong("code", 0L).putBoolean("ready", false).apply()
                return mapOf("state" to "bad", "code" to code)
            }
            if (f.exists()) f.delete()
            if (!pf.renameTo(f)) {
                // Another folder or file system: fall back to a copy.
                pf.copyTo(f, overwrite = true)
                pf.delete()
            }
            // Only now forget the download: DownloadManager.remove deletes a finished file still at its old path.
            try { dm(ctx).remove(*ids) } catch (_: Exception) {}
            p.edit().putString("ids", "").apply()
            p.edit().putBoolean("ready", true).apply()
            return mapOf("state" to "ready", "code" to code, "path" to f.absolutePath, "bytes" to f.length(), "total" to f.length())
        }
        val md = MessageDigest.getInstance("SHA-256")
        tmp.outputStream().use { out ->
            val buf = ByteArray(256 * 1024)
            for (i in 0 until parts) {
                // DownloadManager may pick another name if one was taken; it reports where it really wrote.
                val pf = ids.getOrNull(i)?.let { local[it] }?.takeIf { it.isFile } ?: partFile(ctx, code, i)
                if (!pf.isFile) { tmp.delete(); return mapOf("state" to "failed", "code" to code, "reason" to -1) }
                pf.inputStream().use { inp ->
                    while (true) {
                        val n = inp.read(buf)
                        if (n < 0) break
                        md.update(buf, 0, n)
                        out.write(buf, 0, n)
                    }
                }
            }
        }
        val got = md.digest().joinToString("") { "%02x".format(it) }
        for (i in 0 until parts) partFile(ctx, code, i).delete()
        for (lf in local.values) if (lf.canonicalPath.startsWith(dir(ctx).canonicalPath + File.separator)) lf.delete()
        try { dm(ctx).remove(*ids) } catch (_: Exception) {}
        p.edit().putString("ids", "").apply()
        if (got != sha) {
            tmp.delete()
            p.edit().putLong("code", 0L).putBoolean("ready", false).apply()
            return mapOf("state" to "bad", "code" to code)
        }
        if (f.exists()) f.delete()
        tmp.renameTo(f)
        p.edit().putBoolean("ready", true).apply()
        return mapOf("state" to "ready", "code" to code, "path" to f.absolutePath, "bytes" to f.length(), "total" to f.length())
    }

    /** Stops the download and removes every update file (also used once the app is up to date). */
    fun cancel(ctx: Context) = synchronized(lock) {
        val ids = ids(ctx)
        if (ids.isNotEmpty()) try { dm(ctx).remove(*ids) } catch (_: Exception) {}
        dir(ctx).listFiles()?.forEach { it.delete() }
        prefs(ctx).edit().clear().apply()
    }

    private fun sha256(f: File): String {
        val md = MessageDigest.getInstance("SHA-256")
        f.inputStream().use { inp ->
            val buf = ByteArray(256 * 1024)
            while (true) {
                val n = inp.read(buf)
                if (n < 0) break
                md.update(buf, 0, n)
            }
        }
        return md.digest().joinToString("") { "%02x".format(it) }
    }

    /** "started" | "needs_permission" | "not_ready" | "bad". Checks the file again right before opening the installer. */
    private fun install(ctx: Context): String {
        val p = prefs(ctx)
        val code = p.getLong("code", 0L)
        val f = finalFile(ctx, code)
        if (code <= 0L || !p.getBoolean("ready", false) || !f.isFile) return "not_ready"
        if (sha256(f) != (p.getString("sha", "") ?: "")) {
            cancel(ctx)
            return "bad"
        }
        if (!canInstall(ctx)) return "needs_permission"
        val uri = Uri.parse("content://" + UpdateFiles.authority(ctx) + "/" + f.name)
        val i = Intent(Intent.ACTION_VIEW)
            .setDataAndType(uri, "application/vnd.android.package-archive")
            .addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION or Intent.FLAG_ACTIVITY_NEW_TASK)
        val act = Bridge.activity?.get()
        main.post { if (act != null) act.startActivity(i) else ctx.startActivity(i) }
        return "started"
    }

    private fun openInstallSettings(ctx: Context) {
        val i = if (Build.VERSION.SDK_INT >= 26)
            Intent(Settings.ACTION_MANAGE_UNKNOWN_APP_SOURCES, Uri.parse("package:" + ctx.packageName))
        else Intent(Settings.ACTION_SECURITY_SETTINGS)
        i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        val act = Bridge.activity?.get()
        try { if (act != null) act.startActivity(i) else ctx.startActivity(i) } catch (_: Exception) {
            ctx.startActivity(Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS, Uri.parse("package:" + ctx.packageName)).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK))
        }
    }

    /** From [UpdateDone]: when the last part lands, join it and tell the user it is ready (once). */
    fun onDownloadDone(ctx: Context, id: Long) {
        if (id !in ids(ctx)) return
        val s = status(ctx)
        val p = prefs(ctx)
        if (s["state"] == "ready" && !p.getBoolean("notified", false)) {
            p.edit().putBoolean("notified", true).apply()
            Notifier.task(ctx, 7401, "Your SubMac update is ready. Tap to install.", "", "update", "info")
        }
    }
}

/** DownloadManager tells the app a part finished, even when SubMac is closed. */
class UpdateDone : BroadcastReceiver() {
    override fun onReceive(context: Context, intent: Intent) {
        if (intent.action != DownloadManager.ACTION_DOWNLOAD_COMPLETE) return
        val id = intent.getLongExtra(DownloadManager.EXTRA_DOWNLOAD_ID, -1L)
        if (id < 0) return
        val pr = goAsync()
        Thread {
            try { AppUpdater.onDownloadDone(context.applicationContext, id) } catch (_: Exception) {} finally { pr.finish() }
        }.start()
    }
}

/**
 * Read-only access to the joined update file for Android's installer (files/updates only). Not exported: the
 * installer gets a one-time read grant from the install intent. Same idea as androidx FileProvider, without the
 * extra dependency.
 */
class UpdateFiles : ContentProvider() {
    companion object {
        fun authority(ctx: Context) = ctx.packageName + ".updates"
    }

    private fun fileFor(uri: Uri): File {
        val name = uri.lastPathSegment ?: throw FileNotFoundException()
        val ctx = context ?: throw FileNotFoundException()
        val dir = File(ctx.getExternalFilesDir(null) ?: ctx.filesDir, "updates")
        val f = File(dir, name)
        if (!name.endsWith(".apk") || !f.canonicalPath.startsWith(dir.canonicalPath + File.separator) || !f.isFile) throw FileNotFoundException()
        return f
    }

    override fun onCreate() = true

    override fun openFile(uri: Uri, mode: String): android.os.ParcelFileDescriptor =
        android.os.ParcelFileDescriptor.open(fileFor(uri), android.os.ParcelFileDescriptor.MODE_READ_ONLY)

    override fun getType(uri: Uri): String = "application/vnd.android.package-archive"

    override fun query(uri: Uri, projection: Array<out String>?, selection: String?, selectionArgs: Array<out String>?, sortOrder: String?): Cursor {
        val f = fileFor(uri)
        val cols = projection ?: arrayOf(OpenableColumns.DISPLAY_NAME, OpenableColumns.SIZE)
        val c = MatrixCursor(cols)
        c.addRow(cols.map { if (it == OpenableColumns.SIZE) f.length() else if (it == OpenableColumns.DISPLAY_NAME) f.name else null }.toTypedArray())
        return c
    }

    override fun insert(uri: Uri, values: ContentValues?): Uri? = null
    override fun delete(uri: Uri, selection: String?, selectionArgs: Array<out String>?) = 0
    override fun update(uri: Uri, values: ContentValues?, selection: String?, selectionArgs: Array<out String>?) = 0
}
