/// In-app updates (branch feature/app-update). Checks `latest_release` on cold start, on resume (at most every
/// 30 minutes) and from Settings > App update. The download itself runs in Android's DownloadManager
/// (android_native/AppUpdater.kt), so it keeps going when the user leaves SubMac. State is saved, so a restart
/// shows the same window: progress, or Install straight away when the file is already here.
class Updater extends ChangeNotifier {
  Updater._();
  static final Updater I = Updater._();

  static const _ch = MethodChannel('submac/update');
  static const _key = 'app_update_v1';
  static const resumeEvery = Duration(minutes: 30);

  bool supported = false;
  int? currentCode;
  String currentName = AppConfig.version;
  Release? release;
  final UpdateMachine machine = UpdateMachine();
  bool dismissed = false;
  bool checking = false;
  bool checkFailed = false;
  DateTime? lastCheck;
  DateTime? _fetchedAt;
  bool _pendingInstall = false;
  bool _started = false;
  Timer? _poll;
  Timer? _tick;
  bool _lastForced = false;

  UpdateState get state => machine.state;
  UpdatePhase get phase => machine.state.phase;

  DateTime get now => effectiveNow(serverNow: release?.serverNow, fetchedAt: _fetchedAt, deviceNow: DateTime.now().toUtc());
  bool get forced => release != null && isForced(release!.requiredAt, now);
  bool get pending => release != null && release!.newerThan(currentCode);
  bool get showWindow => shouldShowWindow(release: release, currentCode: currentCode, forced: forced, dismissed: dismissed);

  /// Once per process, right after runApp.
  Future<void> init() async {
    if (_started) return;
    _started = true;
    try {
      final info = await _ch.invokeMapMethod<String, dynamic>('info');
      currentCode = (info?['code'] as num?)?.toInt();
      final n = info?['name'];
      if (n is String && n.isNotEmpty) currentName = n;
      supported = currentCode != null;
    } catch (_) {
      supported = false; // not Android (or tests): no updates
    }
    if (!supported) return;
    _ch.setMethodCallHandler((call) async {
      // The SubMac screen was opened again (the app kept running in the background): show the window again.
      if (call.method == 'opened') {
        dismissed = false;
        notifyListeners();
        unawaited(resumed(force: true));
      }
      return null;
    });
    await _load();
    if (release != null && !release!.newerThan(currentCode)) {
      // Updated since last time: remove the file and forget the release.
      await _clearFiles();
      release = null;
      machine.onRelease(null, currentCode);
      await _save();
    }
    notifyListeners();
    await _syncNative();
    unawaited(check());
    _tick = Timer.periodic(const Duration(seconds: 30), (_) {
      // A deadline passing while the app is open turns the window into the full-screen block.
      if (forced != _lastForced) {
        _lastForced = forced;
        notifyListeners();
      }
    });
  }

  /// From the app's resume hook. Checks the server at most every 30 minutes unless [force].
  Future<void> resumed({bool force = false}) async {
    if (!supported) return;
    machine.onResumed();
    try {
      final info = await _ch.invokeMapMethod<String, dynamic>('info');
      final can = info?['canInstall'] == true;
      final code = (info?['code'] as num?)?.toInt();
      if (code != null) currentCode = code;
      if (_pendingInstall && can) {
        _pendingInstall = false;
        await _syncNative();
        if (phase == UpdatePhase.downloaded) unawaited(install());
      }
    } catch (_) {}
    await _syncNative();
    notifyListeners();
    final last = lastCheck;
    if (force || last == null || DateTime.now().difference(last) >= resumeEvery) await check();
  }

  /// Asks the server for the newest published release above this build.
  Future<void> check({bool manual = false}) async {
    if (!supported || checking) return;
    checking = true;
    checkFailed = false;
    if (manual) notifyListeners();
    try {
      final r = await Supabase.instance.client.rpc('latest_release', params: {'p_current_code': currentCode ?? 0}).timeout(const Duration(seconds: 15));
      final rel = Release.fromRpc(r);
      lastCheck = DateTime.now();
      _fetchedAt = DateTime.now().toUtc();
      final serverNow = r is Map ? DateTime.tryParse('${r['server_now']}')?.toUtc() : null;
      final cleanup = machine.onRelease(rel, currentCode);
      if (cleanup) await _clearFiles();
      if (rel == null) {
        release = null;
      } else {
        release = Release.fromRpc({...rel.toJson(), 'server_now': (serverNow ?? rel.serverNow)?.toIso8601String()});
      }
      _lastForced = forced;
      await _save();
      await _syncNative();
    } catch (e) {
      checkFailed = true;
      debugPrint('update check failed: $e');
    } finally {
      checking = false;
      notifyListeners();
    }
  }

  Future<void> download() async {
    final r = release;
    if (r == null || !supported) return;
    if (phase != UpdatePhase.idle && phase != UpdatePhase.cancelled) return;
    machine.start();
    notifyListeners();
    try {
      await _ch.invokeMethod('start', {
        'code': r.versionCode,
        'urls': r.urls(AppConfig.supabaseUrl),
        'sha256': r.sha256,
        'size': r.sizeBytes,
        'title': 'SubMac ${r.versionName}',
      });
    } catch (e) {
      debugPrint('update start failed: $e');
      machine.onNative('failed');
    }
    await _save();
    _watch();
    notifyListeners();
  }

  /// Stops the download. The window comes back so the user can download again any time.
  Future<void> cancel() async {
    try {
      await _ch.invokeMethod('cancel');
    } catch (_) {}
    machine.cancel();
    dismissed = false;
    await _save();
    _watch();
    notifyListeners();
  }

  Future<void> install() async {
    if (phase != UpdatePhase.downloaded) return;
    machine.install();
    notifyListeners();
    String r;
    try {
      r = await _ch.invokeMethod<String>('install') ?? 'not_ready';
    } catch (_) {
      r = 'not_ready';
    }
    machine.onInstallResult(r);
    if (r == 'needs_permission') _pendingInstall = true;
    await _save();
    notifyListeners();
  }

  /// Opens Android's "Install unknown apps" switch for SubMac; Install continues when the user comes back.
  Future<void> openInstallSettings() async {
    _pendingInstall = true;
    try {
      await _ch.invokeMethod('openInstallSettings');
    } catch (_) {}
  }

  /// Later: hides the window until the next start (never for a required update).
  void later() {
    if (forced) return;
    dismissed = true;
    notifyListeners();
  }

  // ---------------- internals ----------------

  Future<void> _syncNative() async {
    if (!supported || state.releaseCode == 0) return;
    try {
      final s = await _ch.invokeMapMethod<String, dynamic>('status');
      final st = '${s?['state'] ?? 'none'}';
      final code = (s?['code'] as num?)?.toInt() ?? 0;
      if (code != 0 && code != state.releaseCode && st != 'none') {
        // Files of another release: remove them.
        await _clearFiles();
        return;
      }
      final before = phase;
      machine.onNative(st, bytes: (s?['bytes'] as num?)?.toInt() ?? 0, total: (s?['total'] as num?)?.toInt() ?? 0);
      // Back to the window when the download finished or stopped while it was hidden.
      if (before == UpdatePhase.downloading && (phase == UpdatePhase.downloaded || phase == UpdatePhase.cancelled)) dismissed = false;
      await _save();
    } catch (_) {}
    _watch();
  }

  void _watch() {
    if (phase == UpdatePhase.downloading) {
      _poll ??= Timer.periodic(const Duration(seconds: 1), (_) async {
        await _syncNative();
        notifyListeners();
      });
    } else {
      _poll?.cancel();
      _poll = null;
    }
  }

  Future<void> _clearFiles() async {
    try {
      await _ch.invokeMethod('cancel');
    } catch (_) {}
  }

  Future<void> _load() async {
    try {
      final p = await SharedPreferences.getInstance();
      final raw = p.getString(_key);
      if (raw == null) return;
      final j = jsonDecode(raw);
      if (j is! Map) return;
      release = Release.fromRpc(j['release']);
      machine.state = UpdateState.fromJson(j['state']);
      final f = DateTime.tryParse('${j['fetched_at']}');
      _fetchedAt = f?.toUtc();
      if (release == null) machine.state = const UpdateState();
    } catch (_) {}
  }

  Future<void> _save() async {
    try {
      final p = await SharedPreferences.getInstance();
      if (release == null) {
        await p.remove(_key);
        return;
      }
      await p.setString(_key, jsonEncode({'release': release!.toJson(), 'state': state.toJson(), 'fetched_at': _fetchedAt?.toIso8601String()}));
    } catch (_) {}
  }

  @override
  void dispose() {
    _poll?.cancel();
    _tick?.cancel();
    super.dispose();
  }
}

/// Pure logic of in-app updates (no Flutter, no platform): version compare, forced vs optional by time,
/// the download state machine and when the update window shows. The controller is `updater.dart`.
library;

/// Compares "1.13.50+83" style versions: dot parts as numbers, then the build after '+'. Missing parts count as 0.
int compareVersions(String a, String b) {
  List<int> nums(String s) => s.split(RegExp(r'[.+]')).map((p) => int.tryParse(RegExp(r'\d+').stringMatch(p) ?? '') ?? 0).toList();
  String core(String s) => s.trim().split('+').first;
  String build(String s) => s.contains('+') ? s.split('+').last : '0';
  final x = nums(core(a)), y = nums(core(b));
  for (var i = 0; i < (x.length > y.length ? x.length : y.length); i++) {
    final p = i < x.length ? x[i] : 0, q = i < y.length ? y[i] : 0;
    if (p != q) return p < q ? -1 : 1;
  }
  final bx = int.tryParse(build(a)) ?? 0, by = int.tryParse(build(b)) ?? 0;
  return bx == by ? 0 : (bx < by ? -1 : 1);
}

/// Required when a deadline exists and has been reached.
bool isForced(DateTime? requiredAt, DateTime now) => requiredAt != null && !now.isBefore(requiredAt);

/// "Now" anchored on the server's clock at the last check, moved forward by the time since then, so changing the
/// phone's clock does not skip a required update. Time never runs backwards from the server's value.
DateTime effectiveNow({DateTime? serverNow, DateTime? fetchedAt, required DateTime deviceNow}) {
  if (serverNow == null || fetchedAt == null) return deviceNow;
  final elapsed = deviceNow.difference(fetchedAt);
  return serverNow.add(elapsed.isNegative ? Duration.zero : elapsed);
}

class ReleasePart {
  final String path;
  final int size;
  const ReleasePart(this.path, this.size);
  Map<String, dynamic> toJson() => {'path': path, 'size': size};
}

/// One published release, as `latest_release(p_current_code)` returns it.
class Release {
  final String id;
  final int versionCode;
  final String versionName;
  final String notes;
  final String bucket;

  /// 'r2' = one file at [fileUrl] (Cloudflare R2); 'supabase' = [parts] in the Supabase bucket, joined on the phone.
  final String storage;
  final String? fileUrl;
  final List<ReleasePart> parts;
  final int sizeBytes;
  final String sha256;
  final DateTime? requiredAt;
  final DateTime? serverNow;

  const Release({
    required this.id,
    required this.versionCode,
    required this.versionName,
    this.notes = '',
    this.bucket = 'app-updates',
    this.storage = 'supabase',
    this.fileUrl,
    this.parts = const [],
    this.sizeBytes = 0,
    required this.sha256,
    this.requiredAt,
    this.serverNow,
  });

  static DateTime? _date(dynamic v) => v == null ? null : DateTime.tryParse(v.toString())?.toUtc();

  /// null when there is no update (or the answer is not usable).
  static Release? fromRpc(dynamic j) {
    if (j is! Map || j['update'] != true) return null;
    final code = j['version_code'] is num ? (j['version_code'] as num).toInt() : int.tryParse('${j['version_code']}');
    final sha = '${j['sha256'] ?? ''}'.toLowerCase();
    final parts = <ReleasePart>[
      for (final p in (j['parts'] is List ? j['parts'] as List : const []))
        if (p is Map && p['path'] is String) ReleasePart(p['path'] as String, p['size'] is num ? (p['size'] as num).toInt() : 0),
    ];
    final storage = '${j['storage'] ?? 'supabase'}';
    final rawUrl = j['file_url'];
    final fileUrl = rawUrl is String && rawUrl.startsWith('https://') ? rawUrl : null;
    if (code == null || code <= 0 || !RegExp(r'^[0-9a-f]{64}$').hasMatch(sha)) return null;
    if (storage == 'r2' ? fileUrl == null : parts.isEmpty) return null;
    return Release(
      id: '${j['id'] ?? ''}',
      versionCode: code,
      versionName: '${j['version_name'] ?? ''}',
      notes: '${j['notes'] ?? ''}',
      bucket: '${j['bucket'] ?? 'app-updates'}',
      storage: storage == 'r2' ? 'r2' : 'supabase',
      fileUrl: storage == 'r2' ? fileUrl : null,
      parts: parts,
      sizeBytes: j['size_bytes'] is num ? (j['size_bytes'] as num).toInt() : parts.fold(0, (a, p) => a + p.size),
      sha256: sha,
      requiredAt: _date(j['required_at']),
      serverNow: _date(j['server_now']),
    );
  }

  Map<String, dynamic> toJson() => {
        'update': true,
        'id': id,
        'version_code': versionCode,
        'version_name': versionName,
        'notes': notes,
        'bucket': bucket,
        'storage': storage,
        'file_url': fileUrl,
        'parts': [for (final p in parts) p.toJson()],
        'size_bytes': sizeBytes,
        'sha256': sha256,
        'required_at': requiredAt?.toIso8601String(),
        'server_now': serverNow?.toIso8601String(),
      };

  /// True when the phone downloads one file (R2) instead of joining parts.
  bool get singleFile => storage == 'r2' && fileUrl != null;

  /// What to download, in order: the one R2 file, or the public address of every Supabase part.
  List<String> urls(String supabaseUrl) => singleFile
      ? [fileUrl!]
      : [
          for (final p in parts) '${supabaseUrl.replaceAll(RegExp(r'/+$'), '')}/storage/v1/object/public/$bucket/${p.path}',
        ];

  bool newerThan(int? currentCode) => currentCode != null && versionCode > currentCode;
}

/// What the update window is doing.
enum UpdatePhase { idle, downloading, cancelled, downloaded, installing }

/// Why the last step did not go through (shown as one plain line).
enum UpdateIssue { none, failed, badFile, needsPermission }

class UpdateState {
  final UpdatePhase phase;
  final int releaseCode; // 0 = no release
  final int bytes;
  final int total;
  final UpdateIssue issue;

  const UpdateState({this.phase = UpdatePhase.idle, this.releaseCode = 0, this.bytes = 0, this.total = 0, this.issue = UpdateIssue.none});

  double? get progress => total > 0 ? (bytes / total).clamp(0.0, 1.0) : null;

  UpdateState copyWith({UpdatePhase? phase, int? releaseCode, int? bytes, int? total, UpdateIssue? issue}) => UpdateState(
        phase: phase ?? this.phase,
        releaseCode: releaseCode ?? this.releaseCode,
        bytes: bytes ?? this.bytes,
        total: total ?? this.total,
        issue: issue ?? this.issue,
      );

  Map<String, dynamic> toJson() => {'phase': phase.name, 'code': releaseCode, 'bytes': bytes, 'total': total, 'issue': issue.name};

  static UpdateState fromJson(dynamic j) {
    if (j is! Map) return const UpdateState();
    T pick<T extends Enum>(List<T> all, dynamic v, T d) => all.firstWhere((e) => e.name == v, orElse: () => d);
    return UpdateState(
      phase: pick(UpdatePhase.values, j['phase'], UpdatePhase.idle),
      releaseCode: j['code'] is num ? (j['code'] as num).toInt() : 0,
      bytes: j['bytes'] is num ? (j['bytes'] as num).toInt() : 0,
      total: j['total'] is num ? (j['total'] as num).toInt() : 0,
      issue: pick(UpdateIssue.values, j['issue'], UpdateIssue.none),
    );
  }
}

/// The download state machine: idle -> downloading -> downloaded -> installing, with cancelled as the way back.
/// Every method returns the next state; illegal moves return the state unchanged.
class UpdateMachine {
  UpdateState state;
  UpdateMachine([this.state = const UpdateState()]);

  /// A (new) answer from the server. Returns true when old update files must be removed.
  bool onRelease(Release? r, int? currentCode) {
    if (r == null || !r.newerThan(currentCode)) {
      final hadFiles = state.releaseCode != 0 || state.phase != UpdatePhase.idle;
      state = const UpdateState();
      return hadFiles;
    }
    if (r.versionCode != state.releaseCode) {
      final hadFiles = state.releaseCode != 0 && state.phase != UpdatePhase.idle;
      state = UpdateState(releaseCode: r.versionCode, total: r.sizeBytes);
      return hadFiles;
    }
    return false;
  }

  UpdateState start() {
    if (state.releaseCode == 0) return state;
    if (state.phase == UpdatePhase.idle || state.phase == UpdatePhase.cancelled) {
      state = state.copyWith(phase: UpdatePhase.downloading, bytes: 0, issue: UpdateIssue.none);
    }
    return state;
  }

  UpdateState cancel() {
    if (state.phase == UpdatePhase.downloading) state = state.copyWith(phase: UpdatePhase.cancelled, bytes: 0, issue: UpdateIssue.none);
    return state;
  }

  /// What Android's download reports: none | running | paused | failed | bad | ready.
  UpdateState onNative(String native, {int bytes = 0, int total = 0}) {
    if (state.releaseCode == 0) return state;
    final t = total > 0 ? total : state.total;
    switch (native) {
      case 'running':
      case 'paused':
        if (state.phase == UpdatePhase.idle || state.phase == UpdatePhase.downloading || state.phase == UpdatePhase.cancelled) {
          state = state.copyWith(phase: UpdatePhase.downloading, bytes: bytes, total: t, issue: UpdateIssue.none);
        }
      case 'ready':
        if (state.phase != UpdatePhase.installing) {
          state = state.copyWith(phase: UpdatePhase.downloaded, bytes: t, total: t, issue: state.issue == UpdateIssue.needsPermission ? state.issue : UpdateIssue.none);
        }
      case 'failed':
        if (state.phase == UpdatePhase.downloading) state = state.copyWith(phase: UpdatePhase.cancelled, bytes: 0, issue: UpdateIssue.failed);
      case 'bad':
        state = state.copyWith(phase: UpdatePhase.cancelled, bytes: 0, issue: UpdateIssue.badFile);
      case 'none':
        // The files are gone (removed from Android's Downloads list, or never started).
        if (state.phase == UpdatePhase.downloading) {
          state = state.copyWith(phase: UpdatePhase.cancelled, bytes: 0, issue: UpdateIssue.failed);
        } else if (state.phase == UpdatePhase.downloaded || state.phase == UpdatePhase.installing) {
          state = state.copyWith(phase: UpdatePhase.idle, bytes: 0, issue: UpdateIssue.none);
        }
    }
    return state;
  }

  UpdateState install() {
    if (state.phase == UpdatePhase.downloaded) state = state.copyWith(phase: UpdatePhase.installing, issue: UpdateIssue.none);
    return state;
  }

  /// What opening the installer returned: started | needs_permission | bad | not_ready.
  UpdateState onInstallResult(String r) {
    if (state.phase != UpdatePhase.installing) return state;
    if (r == 'started') return state;
    if (r == 'needs_permission') {
      state = state.copyWith(phase: UpdatePhase.downloaded, issue: UpdateIssue.needsPermission);
    } else if (r == 'bad') {
      state = state.copyWith(phase: UpdatePhase.cancelled, bytes: 0, issue: UpdateIssue.badFile);
    } else {
      state = state.copyWith(phase: UpdatePhase.cancelled, bytes: 0, issue: UpdateIssue.failed);
    }
    return state;
  }

  /// Back in SubMac with the old version still running: the installer was closed, so Install is offered again.
  UpdateState onResumed() {
    if (state.phase == UpdatePhase.installing) state = state.copyWith(phase: UpdatePhase.downloaded);
    return state;
  }
}

/// The window shows while a newer release is pending, unless the user tapped Later in this run of the app.
/// A required update always shows (and cannot be dismissed).
bool shouldShowWindow({required Release? release, required int? currentCode, required bool forced, required bool dismissed}) {
  if (release == null || !release.newerThan(currentCode)) return false;
  return forced || !dismissed;
}

// CB-20 / CB-45: how the app counts and labels model usage, the same way the server bills it.

import 'llm_client.dart';

/// The weights the server bills with (`quotaOf` in ai-chat): uncached prompt × in, cached prompt × cached, output ×
/// out, rounded up per call. /status may send them; until it does the server's own defaults apply (0.3 / 0.1 / 1).
class UsageUnits {
  static const defaultIn = 0.3, defaultCached = 0.1, defaultOut = 1.0;
  static double wIn = defaultIn, wCached = defaultCached, wOut = defaultOut;

  /// Weights from a /status answer, when it has them (`weights: {input, cached, output}`, or the ai_settings names
  /// `input_weight` / `cached_weight` / `output_weight`, at the top or in `billing`). Anything missing keeps its default.
  static void noteStatus(Object? status) {
    if (status is! Map) return;
    Object? w = status['weights'];
    if (w is! Map && status['billing'] is Map) w = (status['billing'] as Map)['weights'];
    final src = w is Map ? w : (status['billing'] is Map ? status['billing'] as Map : status);
    double pick(List<String> keys, double d) {
      for (final k in keys) {
        final v = src[k];
        final n = v is num ? v.toDouble() : double.tryParse('${v ?? ''}');
        if (n != null && n.isFinite && n >= 0) return n;
      }
      return d;
    }

    wIn = pick(const ['input', 'in', 'input_weight'], defaultIn);
    wCached = pick(const ['cached', 'cache', 'cached_weight'], defaultCached);
    wOut = pick(const ['output', 'out', 'output_weight'], defaultOut);
  }

  static void resetWeights() {
    wIn = defaultIn;
    wCached = defaultCached;
    wOut = defaultOut;
  }

  /// The cached part of a provider's usage block (same fields the server reads).
  static int cachedOf(Object? usage) {
    if (usage is! Map) return 0;
    final details = usage['prompt_tokens_details'], inDetails = usage['input_tokens_details'];
    final candidates = [
      details is Map ? details['cached_tokens'] : null,
      inDetails is Map ? inDetails['cached_tokens'] : null,
      usage['cached_tokens'],
      usage['prompt_cache_hit_tokens'],
      usage['cachedContentTokenCount'],
      usage['cached_content_token_count'],
      usage['cache_read_input_tokens'],
    ];
    for (final c in candidates) {
      if (c is num && c.isFinite && c > 0) return c.floor();
    }
    return 0;
  }

  /// Billing units of one call. [fallback] (an estimate) is used only when the call has no prompt/output counts.
  static int of(int prompt, int output, {int cached = 0, int fallback = 0}) {
    final p = prompt < 0 ? 0 : prompt, c = output < 0 ? 0 : output;
    if (p == 0 && c == 0) return fallback < 0 ? 0 : fallback;
    final hit = cached < 0 ? 0 : (cached > p ? p : cached);
    return ((p - hit) * wIn + hit * wCached + c * wOut).ceil();
  }

  static int ofResult(LlmResult r) => of(r.tokensIn, r.tokensOut, cached: r.tokensCached, fallback: r.tokens);
}

/// The `x-submac-source` of a chat model call (CB-45). Only a bot's own work (a wake, an office job) is bot work;
/// what the owner types to a bot is `bot-chat`, billed to the owner's chat bucket (ai-chat v43, S-2).
class UsageSource {
  static const bot = 'bot', botChat = 'bot-chat', office = 'office', agent = 'agent', side = 'side';

  /// Set once a server refused `bot-chat` (an older ai-chat): owner chats with a bot then go as plain `agent`.
  static bool botChatRefused = false;

  static String pick({required bool officeJob, required bool botConv, required bool botWork, bool sideRun = false}) {
    if (officeJob) return office;
    if (botConv) return botWork ? bot : (botChatRefused ? agent : botChat);
    return sideRun ? side : agent;
  }

  /// Usage the server counts against the bots' share (billing.is_bot): the only usage a bot's daily budget counts.
  static bool isBotWork(String source) => source == bot || source == office || source == 'background' || source == 'bot-cloud';

  /// A 400 that says the source is unknown: retry this call as `agent` and remember it.
  static bool refusedBotChat(String source, LlmException e) {
    if (source != botChat || e.status != 400) return false;
    final s = '${e.code} ${e.message}'.toLowerCase();
    return s.contains('source');
  }
}

/// Which allowance lane an opencode turn's model calls bill to (owner 2026-11-03).
///
/// The cause: opencode on the phone had one provider (`submac`), so every model call it made, a bot's wake and a team
/// member's part included, reached ai-chat without `x-submac-lane: background`. Bot work was billed as the owner's own
/// chat and was never paused when the bots' share ran out; bots also always ran on the `submac` alias.
///
/// The fix: opencode gets one provider per lane, each pointing at its own path on the app's local proxy. The proxy
/// (LocalProxy.kt) turns the path into the lane: `/__submac/lane/<source>/...` goes out with `x-submac-lane: background`
/// and `x-submac-source: <source>`; any other `/__submac/...` call goes out with the lane header removed, so a
/// foreground call can never carry it. Every prompt names its provider and model explicitly (opencode's per-message
/// `model`), so the lane is chosen per turn and per session and cannot leak between turns running side by side: it
/// travels in each request's own path, not in shared state. (opencode would otherwise reuse a session's last model, so
/// a bot chat the owner types in after a background wake would stay on the background lane.) Subagents opencode starts
/// in child sessions inherit the parent message's provider, so they bill to the same lane.
class OcLane {
  OcLane._();

  /// The owner's own chats (and the owner typing in a bot's chat): the user's lane.
  static const fg = 'submac';

  /// A bot's own background work (wakes, routines, follow-ups): the bots' share, billed as `bot`.
  static const bot = 'submac-bg';

  /// Office and team work nobody is watching (team members' and leads' parts): the bots' share, billed as `office`.
  static const office = 'submac-office';

  /// The source each background provider sends (ai-chat BOT_SOURCES: both bill to the bots' share).
  static const bgSources = {bot: 'bot', office: 'office'};

  /// The proxy path prefix that marks the background lane (after LocalProxy.PREFIX).
  static const lanePath = '/lane/';

  /// SubMac's chat model names opencode may be asked for. ai-chat treats every `submac*` chat name as the one model
  /// (older apps sent these aliases), so each is safe to send. The image model is not a chat model.
  static const models = ['submac', 'submac-smart', 'submac-fast', 'submac-think', 'submac-lite', 'submac-vision', 'submac-somali'];

  /// The providers written into opencode's config (OpenCode.kt builds the JSON): id, shown name, proxy path and the
  /// `x-submac-source` header. Every provider offers the same [models].
  static List<Map<String, String>> providers() => [
        {'id': fg, 'name': 'SubMac', 'path': '', 'source': 'opencode'},
        for (final e in bgSources.entries) {'id': e.key, 'name': 'SubMac (${e.value})', 'path': '$lanePath${e.value}', 'source': e.value},
      ];

  /// The SubMac model a bot's turn runs on: its own setting (`auto` or `providerId|model`) when that names one of
  /// SubMac's chat models, else `submac`. A BYOK setting (`byok-…|model`) is resolved first by [pick]; here any other
  /// provider id falls back to `submac`.
  static String modelFor(String? setting) {
    final s = (setting ?? '').trim();
    if (s.isEmpty || s == 'auto') return Cloud.submac;
    final i = s.indexOf('|');
    final provider = i >= 0 ? s.substring(0, i).trim() : '';
    final model = (i >= 0 ? s.substring(i + 1) : s).trim().toLowerCase();
    if (provider.isNotEmpty && provider != Cloud.providerId) return Cloud.submac;
    return models.contains(model) ? model : Cloud.submac;
  }

  /// The prompt's `model` for one turn. [background]: nobody is watching (a bot's wake, office or team work).
  /// [officeWork]: that background work is office/team work. [botModel]: the bot's model setting, null for the owner.
  ///
  /// BYOK (owner 2026-11-03, docs/byok.md): the owner's chats use the model chosen in Settings → Models
  /// ([userModel], default the saved choice), a bot uses its own setting; when either names a usable model of the
  /// user's own providers, the turn goes to that provider and model directly (`byok-…`), whatever the lane: lanes only
  /// decide SubMac billing, and a BYOK call is never billed by SubMac. Anything unusable falls back to SubMac.
  static Map<String, String> pick({required bool background, bool officeWork = false, String? botModel, String? userModel}) {
    final setting = botModel == null ? (userModel ?? ByokStore.I.chatModel) : botSetting(botModel, userModel: userModel);
    final own = byokFor(setting);
    if (own != null) return own;
    return {
      'providerID': background ? (officeWork ? office : bot) : fg,
      'modelID': botModel == null ? Cloud.submac : modelFor(setting),
    };
  }

  /// A bot's model setting left at its default: `auto` or empty ("Default (same as SubMac)" in its profile).
  static bool isDefault(String? setting) {
    final s = (setting ?? '').trim();
    return s.isEmpty || s == 'auto';
  }

  /// The model setting a bot's turn really runs on (owner 2026-11-05: each bot picks its model in its own profile).
  ///  - Default (`auto`): the same model as SubMac, i.e. the owner's chat model ([userModel], default the one chosen
  ///    in Settings → Models), whether that is SubMac or one of the owner's own providers.
  ///  - An own-provider model (`byok-…|model`) that is no longer usable here (provider removed or off, model off,
  ///    no key, or a setting synced from another phone with other providers): falls back to that default.
  ///  - `submac` / `submac|alias`: SubMac's own model, whatever the owner's chats use.
  static String botSetting(String? botModel, {String? userModel, ByokStore? store}) {
    final st = store ?? ByokStore.I;
    final fallback = userModel ?? st.chatModel;
    if (isDefault(botModel)) return fallback;
    final s = botModel!.trim();
    if (ByokRef.parse(s) != null && st.resolve(s) == null) return fallback;
    return s;
  }

  /// Whether a bot's chosen model is set but cannot be used here now (it runs on the default instead).
  static bool botModelUnavailable(String? botModel, {ByokStore? store}) {
    if (isDefault(botModel)) return false;
    return ByokRef.parse(botModel) != null && (store ?? ByokStore.I).resolve(botModel) == null;
  }

  /// The prompt `model` of a BYOK setting, or null when it does not name a usable model of the user's providers.
  static Map<String, String>? byokFor(String? setting, {ByokStore? store}) {
    final pick = (store ?? ByokStore.I).resolve(setting);
    return pick == null ? null : {'providerID': pick.provider.id, 'modelID': pick.model.id};
  }

  /// A turn on one of the user's own providers (not billed by SubMac, not stopped by the bots' share).
  static bool isByok(Map<String, String>? model) => model != null && (model['providerID'] ?? '').startsWith(ByokProvider.prefix);

  static bool isBackground(Map<String, String>? model) => model != null && bgSources.containsKey(model['providerID']);

  /// ai-chat's refusal of background work when the bots' share is used up (429 background_paused). opencode retries
  /// a 429 on its own, so it arrives as a retry status (or an error) carrying the server's message.
  static final _paused = RegExp(r'background_paused|background tasks are paused|paused until (tomorrow|midnight)', caseSensitive: false);
  static bool isPausedText(String? s) => s != null && _paused.hasMatch(s);

  /// ai-chat's refusal once the account's whole daily allowance is used up (foreground and background alike).
  static final _allowance = RegExp(r'allowance|done a lot today|refreshes tomorrow|resting until tomorrow', caseSensitive: false);
  static bool isAllowanceText(String? s) => s != null && _allowance.hasMatch(s);
}

/// Why a background opencode turn must stop now (owner 2026-11-03), checked as its events arrive. Pure, so tested
/// without the engine.
///  - `paused`: ai-chat refused it for the bots' share (opencode would retry the 429 until the turn timed out).
///  - `budget`: the bot's own daily token budget ran out mid-turn ([budgetLeft] tokens were left when it started).
class OcBgGuard {
  OcBgGuard({required this.background, this.budgetLeft, this.byok = false});
  final bool background;
  final int? budgetLeft;

  /// The turn runs on the user's own provider (BYOK, owner 2026-11-03): SubMac's bots' share does not apply to it.
  final bool byok;

  static const paused = 'paused', budget = 'budget';

  /// The account's whole daily allowance is used up: the owner's own chat stops too (it used to retry all evening).
  static const credit = 'credit';

  /// The provider keeps refusing (one call was retried 82 times with nothing to stop it).
  static const stuck = 'stuck';

  /// How long one turn may stay in opencode's own retry state before it is stopped.
  static const maxRetrySpan = Duration(minutes: 3);

  /// Tests: the clock.
  static DateTime Function() clock = DateTime.now;

  DateTime? _retrySince;

  /// Tokens a bot may still use today ([budgetK] thousand a day, 0 = no budget): null when it has no budget.
  static int? leftOf(int budgetK, int usedToday) {
    if (budgetK <= 0) return null;
    final left = budgetK * 1000 - usedToday;
    return left < 0 ? 0 : left;
  }

  String? check(OpenCodeTurn t) {
    // The bots'-share pause also says "allowance", so it is checked first: it never stops the owner's own chat, and
    // for background work it means paused, not the whole account being out.
    final sharePause = !byok && (OcLane.isPausedText(t.retry) || OcLane.isPausedText(t.error));
    // opencode retries a refused call on its own until the turn's 30 minutes run out (bug audit 2026-11-07):
    //  - the day's allowance is used up: the owner's chat said "retrying" all evening; stop now and say so.
    //  - the provider keeps failing: one call was retried 82 times; stop after [maxRetrySpan] in retry.
    if (!byok && !background && !sharePause && (OcLane.isAllowanceText(t.retry) || OcLane.isAllowanceText(t.error))) {
      return credit;
    }
    if (t.retry != null && t.retry!.isNotEmpty) {
      _retrySince ??= clock();
      if (clock().difference(_retrySince!) > maxRetrySpan) return stuck;
    } else {
      _retrySince = null;
    }
    if (!background) return null;
    if (sharePause) return paused;
    final left = budgetLeft;
    if (left != null && t.tokens > 0 && t.tokens >= left) return budget;
    return null;
  }
}

import 'dart:convert';

/// SubMac's tools reach opencode through two small MCP tools instead of one per tool (owner, 2026-11-03: "give it as
/// mcp and skill, so it will just use when needed, not every message"). opencode sends every MCP tool's schema with
/// every model call, so the full set (memory, documents, media, apps, connectors, the user's MCP servers...) stays
/// behind [findTool]; opencode looks a tool up when a task needs one and runs it with [runTool]. The `submac` skill
/// says what is there. Pure, so it is tested without the app.
class SubMacGateway {
  SubMacGateway({required this.tools, required this.run, this.redirect});

  /// A tool this caller may call by another name (a bot's `remember` is its own bot_remember): the name it runs as,
  /// or null. The call still runs under its own name (the app maps it).
  final String? Function(String name)? redirect;

  /// opencode's own tools. Called through run_tool they are not found (2026-11-05: run_tool name "webfetch"), so the
  /// answer says to call them directly instead of a bare "no SubMac tool".
  static const engineTools = {
    'bash', 'read', 'write', 'edit', 'glob', 'grep', 'list', 'ls', 'webfetch', 'websearch', 'codesearch', 'todowrite', 'todoread',
    'task', 'skill', 'patch', 'multiedit', 'question',
  };

  /// The full SubMac tool list (OpenAI function form), read fresh on every lookup.
  final List<Map<String, dynamic>> Function() tools;

  /// Runs one SubMac tool by name (through SubMac's own gate and approval card).
  final Future<String> Function(String name, Map<String, dynamic> args) run;

  static const findName = 'find_tools';
  static const runName = 'run_tool';

  static const findTool = {
    'type': 'function',
    'function': {
      'name': findName,
      'description': 'Find SubMac app tools on this phone (the built-in browser for websites, phone control of other apps, '
          'alarms, SMS, contacts, calendar, WhatsApp, connected apps and their actions, the user\'s MCP servers, '
          'connectors, memory, reading and making documents, pictures and media, the app\'s settings, projects and pages). '
          'Returns matching tools with their parameters; run one with run_tool. An empty query lists every tool by name.',
      'parameters': {
        'type': 'object',
        'properties': {
          'query': {'type': 'string', 'description': 'Words for what you need, e.g. "gmail send", "pdf", "remember"'},
        },
      },
    },
  };

  static const runTool = {
    'type': 'function',
    'function': {
      'name': runName,
      'description': 'Run a SubMac app tool found with find_tools. A risky action asks the user on SubMac\'s approval card first.',
      'parameters': {
        'type': 'object',
        'properties': {
          'name': {'type': 'string', 'description': 'The tool name from find_tools'},
          'arguments': {'type': 'object', 'description': 'The tool\'s arguments, as its parameters describe'},
        },
        'required': ['name'],
      },
    },
  };

  static const definitions = [findTool, runTool];

  /// SubMac tools offered to opencode by their own name as well (not only through run_tool), because a task needs them
  /// without knowing to look (owner 2026-11-04: "no visual output ability, why?"). Each still runs only when the
  /// calling chat's own tool list has it, and through the same gate as run_tool.
  static const directNames = {'generate_image'};

  /// The MCP tool list: find_tools, run_tool, and each direct tool [available] has right now (generate_image only
  /// while SubMac's server can make pictures).
  static List<Map<String, dynamic>> listing(List<Map<String, dynamic>> available) => [
        ...definitions,
        for (final t in available)
          if (directNames.contains(_fn(t)['name'])) t,
      ];

  static Map<String, dynamic> _fn(Map<String, dynamic> t) =>
      t['function'] is Map ? Map<String, dynamic>.from(t['function'] as Map) : t;

  /// The tool a run_tool call names, and its arguments (for showing the step and routing it to its chat).
  static (String, Map<String, dynamic>) inner(Map<String, dynamic> args) {
    // Models also send the arguments as "parameters" (seen 2026-11-03 in a bot's office wake): read either.
    final a = args['arguments'] ?? args['parameters'];
    Map<String, dynamic> m = const {};
    if (a is Map) {
      m = Map<String, dynamic>.from(a);
    } else if (a is String && a.trim().isNotEmpty) {
      try {
        final d = jsonDecode(a);
        if (d is Map) m = Map<String, dynamic>.from(d);
      } catch (_) {}
    }
    return (bare('${args['name'] ?? ''}'.trim()), m);
  }

  /// A tool name without the MCP wrapper's prefix models often add ("submac_run_tool", "submac_apps_list_tools").
  /// Used both where a call is noted for routing and where it runs, so the two always agree (a prefix stripped on one
  /// side only made the route miss and the call fall back to whatever chat was open: bots leak, 2026-11-03).
  static String bare(String name) => name.startsWith('submac_') ? name.substring(7) : name;

  Future<String> call(String name, Map<String, dynamic> args) async {
    if (name == findName) return find('${args['query'] ?? ''}');
    if (directNames.contains(bare(name))) {
      final n = bare(name);
      if (!tools().any((t) => _fn(t)['name'] == n)) return 'Error: $n is not available here.';
      return run(n, args);
    }
    if (name == runName) {
      // inner() already drops the MCP wrapper's prefix models often add (see bare).
      final (tool, a) = inner(args);
      if (tool.isEmpty) return 'Error: name is required. Use find_tools to see the tools.';
      if (tool == findName || tool == runName) return 'Error: $tool is not a SubMac app tool.';
      if (!tools().any((t) => _fn(t)['name'] == tool)) {
        final to = redirect?.call(tool);
        if (to != null && tools().any((t) => _fn(t)['name'] == to)) return run(tool, a);
        if (engineTools.contains(tool.toLowerCase())) {
          return 'Error: "$tool" is one of your own tools, not a SubMac app tool. Call $tool directly (not through run_tool).';
        }
        return 'Error: no SubMac tool named "$tool". ${find(tool.replaceAll('_', ' '))}';
      }
      return run(tool, a);
    }
    return 'Error: unknown tool $name.';
  }

  /// Matching tools with their parameters (best first, at most 8); every name when the query is empty.
  String find(String query) {
    final all = [for (final t in tools()) _fn(t)];
    final words = query.toLowerCase().split(RegExp(r'[^a-z0-9]+')).where((w) => w.length > 1).toList();
    if (words.isEmpty) {
      return 'SubMac tools (look one up with a query for its parameters):\n'
          '${all.map((f) => '- ${f['name']}: ${_short('${f['description'] ?? ''}')}').join('\n')}';
    }
    final scored = <(int, Map<String, dynamic>)>[];
    for (final f in all) {
      final n = '${f['name']}'.toLowerCase();
      final d = '${f['description'] ?? ''}'.toLowerCase();
      var s = 0;
      for (final w in words) {
        if (n.contains(w)) s += 3;
        if (d.contains(w)) s += 1;
      }
      if (s > 0) scored.add((s, f));
    }
    scored.sort((a, b) => b.$1.compareTo(a.$1));
    if (scored.isEmpty) return 'No SubMac tool matches "$query". Try other words, or an empty query for the full list.';
    return jsonEncode([
      for (final (_, f) in scored.take(8))
        {'name': f['name'], 'description': f['description'], 'parameters': f['parameters'] ?? {'type': 'object'}},
    ]);
  }

  static String _short(String d) {
    final i = d.indexOf('. ');
    final s = i > 0 ? d.substring(0, i + 1) : d;
    return s.length > 140 ? '${s.substring(0, 137)}...' : s;
  }

  /// The `submac` opencode skill: what the app offers and how to reach it. opencode sees only its name and description
  /// until it loads it.
  static const skill = '''---
name: submac
description: "SubMac app tools on this phone: a real browser for websites (sign in, forms, shopping, booking), control of the phone and its other apps (tap, type, scroll, alarms, SMS, contacts, calendar, WhatsApp), connected apps (Gmail, Calendar, GitHub...), the user's MCP servers, connectors, memory, reading PDFs/Word/photos, making documents and media, pictures, video, speech and music with the user's own media providers, how to show images, charts, drawings, pages and videos in the chat, the app's settings, projects and pages. Load when a task needs the phone's apps or the user's accounts, or a visual answer."
---

  return context.read<AppState>();
}

void _push(BuildContext context, Widget page) => Navigator.push(context, MaterialPageRoute(builder: (_) => page));

Future<bool> _confirm(BuildContext context, String title, String body, String action, {bool danger = false}) async {
  final ok = await showDialog<bool>(
    context: context,
    builder: (d) => AlertDialog(
      title: Text(title),
      content: Text(body),
      actions: [
        TextButton(onPressed: () => Navigator.pop(d, false), child: Text(tr('Cancel'))),
        TextButton(
          onPressed: () => Navigator.pop(d, true),
          child: Text(action, style: TextStyle(color: danger ? Palette.of(context).accent : null)),
        ),
      ],
    ),
  );
  return ok == true;
}

/// Files SubMac moved to its Trash, with one-tap restore.
class TrashScreen extends StatefulWidget {
  final bool embedded;
  const TrashScreen({super.key, this.embedded = false});

  @override
  State<TrashScreen> createState() => _TrashScreenState();
}

class _TrashScreenState extends State<TrashScreen> {
  List<FileSystemEntity> _items = [];
  bool _loading = true;

  @override
  void initState() {
    super.initState();
    _load();
  }

  Future<void> _load() async {
    final dir = Directory(await AgentTools.trashDir());
    final all = await dir.list().toList();
    final items = all.where((e) => !e.path.endsWith('.origin')).toList()
      ..sort((a, b) => b.path.compareTo(a.path));
    if (mounted) {
      setState(() {
        _items = items;
        _loading = false;
      });
    }
  }

  String _name(FileSystemEntity e) {
    final n = e.path.split('/').last;
    final i = n.indexOf('__');
    return i >= 0 ? n.substring(i + 2) : n;
  }

  Future<void> _restore(FileSystemEntity e) async {
    final origin = File('${e.path}.origin');
    if (!await origin.exists()) return;
    final target = (await origin.readAsString()).trim();
    try {
      await Directory(target.substring(0, target.lastIndexOf('/'))).create(recursive: true);
      try {
        await e.rename(target);
      } catch (_) {
        final r = await Process.run('/system/bin/sh', ['-c', 'cp -r "${e.path}" "$target" && rm -rf "${e.path}"']);
        if (r.exitCode != 0) throw Exception(r.stderr);
      }
      await origin.delete();
      if (mounted) ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Restored ${_name(e)}')));
    } catch (err) {
      if (mounted) ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: const Text('That file could not be put back. It may have been moved or the phone is low on space.')));
    }
    _load();
  }

  Future<void> _empty() async {
    final ok = await showDialog<bool>(
      context: context,
      builder: (d) => AlertDialog(
        title: const Text('Empty the Trash?'),
        content: const Text('These files will be gone for good.'),
        actions: [
          TextButton(onPressed: () => Navigator.pop(d, false), child: Text(tr('Cancel'))),
          TextButton(
            onPressed: () => Navigator.pop(d, true),
            child: Text('Empty', style: TextStyle(color: Palette.of(context).accent)),
          ),
        ],
      ),
    );
    if (ok != true) return;
    final dir = Directory(await AgentTools.trashDir());
    await dir.delete(recursive: true);
    _load();
  }

  @override
  Widget build(BuildContext context) {
    final p = Palette.of(context);
    return maybeScaffold(
      embedded: widget.embedded,
      appBar: AppBar(
        title: Text(tr('Trash'), style: Brand.serif(size: 28, color: p.text)),
        actions: [if (_items.isNotEmpty) TextButton(onPressed: _empty, child: const Text('Empty'))],
      ),
      body: _loading
          ? const Center(child: RedDot(size: 10, animate: true))
          : _items.isEmpty
              ? const Center(child: EmptyHint(text: 'The Trash is empty.'))
              : ListView(
                  children: [
                    // Embedded in Files there is no app bar, so Empty sits at the top of the list.
                    if (widget.embedded)
                      Align(
                        alignment: AlignmentDirectional.centerEnd,
                        child: TextButton(onPressed: _empty, child: const Text('Empty')),
                      ),
                    for (final e in _items)
                      ListTile(
                        title: Text(_name(e), maxLines: 1, overflow: TextOverflow.ellipsis),
                        subtitle: Text(e is Directory ? 'Folder' : 'File'),
                        trailing: TextButton(onPressed: () => _restore(e), child: Text(tr('Restore'))),
                      ),
                  ],
                ),
    );
  }
}

String _mb(int bytes) => '${(bytes / 1048576).toStringAsFixed(1)} MB';

String _when(DateTime utc) {
  final d = utc.toLocal();
  const m = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
  String two(int x) => x.toString().padLeft(2, '0');
  return '${d.day} ${m[d.month - 1]} ${d.year}, ${two(d.hour)}:${two(d.minute)}';
}

          _UpdSection(t.whatsNew, [
            Padding(
              padding: const EdgeInsetsDirectional.fromSTEB(16, 14, 16, 16),
              child: Text(r.notes.trim(),
                  key: const ValueKey('upd_notes'), softWrap: true, style: TextStyle(color: p.text, fontSize: AppText.body, height: 1.5)),
            ),
          ]),
        if (r != null)
          _UpdSection(t.details, [
            _UpdRow(t.version(r.versionName), _mb(r.sizeBytes)),
            if (req != null) _UpdRow(t.requiredFrom(_when(req)), ''),
          ]),
      ]),
    );
  }
}

class _UpdCard extends StatelessWidget {
  final List<Widget> children;
  const _UpdCard({required this.children});

  @override
  Widget build(BuildContext context) {
    final p = Palette.of(context);
    return Container(
      decoration: BoxDecoration(
        color: p.sheet,
        borderRadius: Radii.fieldAll,
        border: Border.all(color: p.border, width: 0.8),
      ),
      clipBehavior: Clip.antiAlias,
      child: Material(
        color: Colors.transparent,
        child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
          for (var i = 0; i < children.length; i++) ...[
            if (i > 0) Divider(height: 1, indent: 18, color: p.border),
            children[i],
          ],
        ]),
      ),
    );
  }
}

/// Same look as the Settings sections: a small caps title over a rounded card.
class _UpdSection extends StatelessWidget {
  final String title;
  final List<Widget> children;
  const _UpdSection(this.title, this.children);

  @override
  Widget build(BuildContext context) {
    final p = Palette.of(context);
    return Padding(
      padding: const EdgeInsetsDirectional.fromSTEB(16, 10, 16, 8),
      child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
        Padding(
          padding: const EdgeInsetsDirectional.only(start: 4, bottom: 8),
          child: Text(title.toUpperCase(),
              style: TextStyle(color: p.muted, fontSize: AppText.caption, fontWeight: FontWeight.w600, letterSpacing: 0.8)),
        ),
        _UpdCard(children: children),
      ]),
    );
  }
}

class _UpdRow extends StatelessWidget {
  final String label;
  final String value;
  const _UpdRow(this.label, this.value);

  @override
  Widget build(BuildContext context) {
    final p = Palette.of(context);
    return Padding(
      padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 13),
      child: Row(crossAxisAlignment: CrossAxisAlignment.start, children: [
        Expanded(child: Text(label, softWrap: true, style: TextStyle(color: p.text, fontSize: AppText.body, height: 1.3))),
        if (value.isNotEmpty) ...[
          const SizedBox(width: 12),
          Text(value, style: TextStyle(color: p.muted, fontSize: AppText.body)),
        ],
      ]),
    );
  }
}
