6:/// Which Telesia chat (or team member) a Telesia tool call from opencode belongs to.
7:///
8:/// An MCP call from opencode does not say which session made it. The session's own event stream does: a tool part
9:/// with the tool's name and arguments appears there as the call starts. Each such part is noted here with its chat,
10:/// and the MCP call takes the entry with the same name and the same arguments. Matching on the arguments (not only the
11:/// name) keeps two chats, two bots or two team members that run the same tool at once apart, so one bot's memory,
12:/// desk or approval never lands on another (no leaks between chats).
13:///
14:/// Entries are taken once and expire, so the table never grows. Pure, so it is tested without the app.
15:class McpRoutes {
16:  McpRoutes({this.ttl = const Duration(minutes: 10), this.max = 200, DateTime Function()? clock}) : _clock = clock ?? DateTime.now;
17:
18:  final Duration ttl;
19:  final int max;
20:  final DateTime Function() _clock;
21:  final List<({String key, String tool, String conv, DateTime at, String part})> _open = [];
22:  final Set<String> _seenParts = {};
23:
24:  /// Arguments in a fixed key order, so the same call matches however its map was built.
25:  static String canon(Object? v) {
26:    if (v is Map) {
27:      final keys = v.keys.map((k) => '$k').toList()..sort();
28:      return '{${keys.map((k) => '${jsonEncode(k)}:${canon(v[k])}').join(',')}}';
29:    }
30:    if (v is List) return '[${v.map(canon).join(',')}]';
31:    if (v is num && v == v.roundToDouble() && v.abs() < 1e15) return '${v.toInt()}';
32:    return jsonEncode(v);
33:  }
34:
35:  static String keyOf(String tool, Map<String, dynamic> args) => '$tool ${canon(args)}';
36:
37:  void _prune() {
38:    final cut = _clock().subtract(ttl);
39:    _open.removeWhere((e) => e.at.isBefore(cut));
40:    while (_open.length > max) {
41:      _open.removeAt(0);
42:    }
43:    if (_seenParts.length > max * 4) _seenParts.clear();
44:  }
45:
46:  /// A tool part seen in [conv]'s event stream. [part] is the part's id: a part is noted once however many updates
47:  /// it gets.
48:  void note(String tool, Map<String, dynamic> args, String conv, {required String part}) {
49:    if (tool.isEmpty || !_seenParts.add(part)) return;
50:    _prune();
51:    _open.add((key: keyOf(tool, args), tool: tool, conv: conv, at: _clock(), part: part));
52:  }
53:
54:  /// The chat of an MCP call. The event may arrive a moment after the call, so this waits up to [wait] for it. Null:
55:  /// no matching part (the caller falls back to the open chat).
56:  ///
57:  /// [fuzzy]: when no entry has the same arguments, take the only open entry of that tool. Only safe while one turn
58:  /// runs: with several, that entry may be another turn's call whose own MCP request has not arrived yet, and taking
59:  /// it would run this call as the other chat (bots team audit 2026-11-04).
60:  ///
61:  /// [orphan] (''): the call came from a session no running turn owns (see [watchOrphans]); the caller refuses it.
62:  Future<String?> take(String tool, Map<String, dynamic> args,
63:      {Duration wait = const Duration(milliseconds: 1500), bool fuzzy = true}) async {
64:    final key = keyOf(tool, args);
65:    final until = _clock().add(wait);
66:    while (true) {
67:      _prune();
68:      final i = _open.indexWhere((e) => e.key == key);
69:      if (i >= 0) return _open.removeAt(i).conv;
70:      if (!_clock().isBefore(until)) break;
71:      await Future<void>.delayed(const Duration(milliseconds: 50));
72:    }
73:    // Arguments that did not survive the trip byte for byte: the only open call of that tool, when there is one.
74:    if (!fuzzy) return null;
75:    final same = _open.where((e) => e.tool == tool).toList();
76:    if (same.length == 1) {
77:      _open.remove(same.first);
78:      return same.first.conv;
79:    }
80:    return null;
81:  }
82:
83:  int get openCount => _open.length;

85:  /// Notes the Telesia gateway calls (find_tools, run_tool) in one engine event for [conv], when the event comes from
86:  /// a session of [family]: the turn's own session and every sub-session opencode starts under it (its `task` tool
87:  /// runs a subagent in a child session). The family grows here as `session.created`/`session.updated` events name a
88:  /// parent in it. Without this, a subagent's Telesia call had no route and ran as whatever chat was open (a bot's
89:  /// work could land in the owner's chat and the other way round; 2026-11-03).
90:  void watch(Map<String, dynamic> e, String conv, Set<String> family) {
91:    final props = e['properties'];
92:    if (props is! Map) return;
93:    final type = e['type'];
94:    if (type == 'session.created' || type == 'session.updated') {
95:      final info = props['info'];
96:      if (info is Map && info['id'] != null && family.contains('${info['parentID']}')) family.add('${info['id']}');
97:      return;
98:    }
99:    if (type != 'message.part.updated') return;
100:    final p = props['part'];
101:    if (p is! Map || p['type'] != 'tool' || !family.contains('${p['sessionID']}')) return;
102:    final st = p['state'] is Map ? p['state'] as Map : const {};
103:    if (st['status'] != 'running' || st['input'] is! Map) return;
104:    final input = Map<String, dynamic>.from(st['input'] as Map);
105:    final tool = TelesiaGateway.bare('${p['tool'] ?? ''}');
106:    if (tool == TelesiaGateway.runName) {
107:      final (n, a) = TelesiaGateway.inner(input);
108:      if (n.isNotEmpty) note(n, a, conv, part: '${p['id']}');
109:    } else if ((tool == TelesiaGateway.findName || TelesiaGateway.directNames.contains(tool)) && '${p['tool']}' != tool) {
110:      note(tool, input, conv, part: '${p['id']}');
111:    }
112:  }

114:  /// The route of a call that came from no running turn's session.
115:  static const orphan = '';
116:
117:  /// Notes Telesia gateway calls made by sessions that no running turn owns ([owned] is false for them): a turn that
118:  /// was stopped while opencode still finished a step, a deleted team session, the opencode test screen. They are
119:  /// noted as [orphan], so their MCP call is refused instead of falling back to "the only turn running", which ran
120:  /// it as that other chat (a leak). Call it before the turns' own [watch] for each event.
121:  void watchOrphans(Map<String, dynamic> e, bool Function(String session) owned) {
122:    if (e['type'] != 'message.part.updated') return;
123:    final props = e['properties'];
124:    final p = props is Map ? props['part'] : null;
125:    if (p is! Map || p['type'] != 'tool') return;
126:    final session = '${p['sessionID'] ?? ''}';
127:    if (session.isEmpty || owned(session)) return;
128:    watch(e, orphan, {session});
129:  }

131:  /// The Telesia chat of [session]: its own, or the nearest parent's ([parent]: child session -> parent, from
132:  /// `session.created` events). A subagent's permission or question card shows in the chat whose turn started it,
133:  /// never in whatever chat happens to be open (2026-11-03).
134:  static String? rootConv(String session, String? Function(String session) convOf, Map<String, String> parent) {
135:    var s = session;
136:    for (var i = 0; i < 16; i++) {
137:      final c = convOf(s);
138:      if (c != null) return c;
139:      final up = parent[s];
140:      if (up == null) return null;
141:      s = up;
142:    }
143:    return null;
144:  }

146:  /// Where a call with no matching part goes: the only Telesia turn running (chat, bot or team member), or null when
147:  /// there are several or none. Never "the chat on screen": that is how one bot's call used to act on another chat.
148:  static String? soleLive(Iterable<String> live) {
149:    final l = live.toSet();
150:    return l.length == 1 ? l.first : null;
151:  }
