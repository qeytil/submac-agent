/// The first-launch intro (owner, 2026-10-26): on a clean white page the wordmark "SubMac" draws itself in behind a
/// thin brand-blue pen line, then resolves into the logo (it scales and fades in), and the white dot lands last.
/// As the pen passes each letter a light selection tick fires with one soft glass-like pluck (rising A-major
/// pentatonic notes), then a warm two-note bloom plays as the logo lands, with one soft impact when the dot touches down. About three seconds; a tap anywhere skips
/// it (sound and ticks stop at once). With Remove animations the finished logo simply fades in, with one light tick.
///
/// Pure Flutter: one AnimationController, every beat a fraction of it (IntroMotion).
class IntroScreen extends StatefulWidget {
  const IntroScreen({super.key, required this.onDone, this.sound});

  /// Called once, when the intro finishes or is skipped.
  final VoidCallback onDone;

  /// The sound to play; the device player by default. Tests pass a fake.
  final IntroSound? sound;

  @override
  State<IntroScreen> createState() => IntroScreenState();
}

class IntroScreenState extends State<IntroScreen> with SingleTickerProviderStateMixin {
  late final AnimationController _c = AnimationController(vsync: this, duration: IntroMotion.total);
  IntroSound? _sound;
  bool _started = false;
  bool _done = false;
  bool _still = false;
  int _nextTick = 0;
  bool _landed = false;
  late final List<double> _ticks = IntroMotion.ticks();

  /// True once the intro has finished or was skipped. For tests.
  bool get done => _done;

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();
    if (_started) return;
    _started = true;
    _still = SubMotion.still(context);
    _c.addStatusListener((s) {
      if (s == AnimationStatus.completed) _finish();
    });
    if (_still) {
      // Remove animations: the finished logo fades in quickly, one light tick, no sound.
      _c.duration = IntroMotion.still;
      unawaited(HapticFeedback.selectionClick());
    } else {
      _c.addListener(_beats);
      _sound = widget.sound ?? IntroSound.device();
      unawaited(_sound!.play().catchError((Object _) {})); // a sound that fails never stops the intro
    }
    _c.forward();
  }

  /// Haptics in step with the picture: a selection tick as the pen passes each letter, one soft impact as the dot lands.
  void _beats() {
    final t = _c.value;
    while (_nextTick < _ticks.length && t >= _ticks[_nextTick]) {
      _nextTick++;
      unawaited(HapticFeedback.selectionClick());
    }
    if (!_landed && t >= IntroMotion.dotLand) {
      _landed = true;
      unawaited(HapticFeedback.mediumImpact());
    }
  }

  /// Tap to skip: sound and ticks stop at once.
  void skip() {
    if (_done) return;
    _c.removeListener(_beats);
    _c.stop();
    unawaited(_sound?.stop().catchError((Object _) {}));
    _finish();
  }

  void _finish() {
    if (_done) return;
    _done = true;
    _c.removeListener(_beats);
    widget.onDone();
  }

  @override
  void dispose() {
    _c.removeListener(_beats);
    _c.dispose();
    // A finished intro lets the sound's short tail ring out; a skipped one has already stopped it.
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Semantics(
      button: true,
      label: tr('Skip intro'),
      child: GestureDetector(
        behavior: HitTestBehavior.opaque,
        onTap: skip,
        // Material gives the wordmark a proper text style (no debug underline) on the plain white page.
        child: Material(
          color: IntroMotion.paper,
          child: Center(
            child: RepaintBoundary(
              child: AnimatedBuilder(
                animation: _c,
                builder: (context, _) => _still ? _stillFrame(_c.value) : _frame(_c.value),
              ),
            ),
          ),
        ),
      ),
    );
  }

  Widget _stillFrame(double t) => Opacity(
        opacity: IntroMotion.span(t, 0, 0.35),
        child: const IntroLogo(dot: 1),
      );

  Widget _frame(double t) {
    final draw = IntroMotion.span(t, IntroMotion.drawStart, IntroMotion.drawEnd);
    final wordOut = IntroMotion.easeInOut.transform(IntroMotion.span(t, IntroMotion.wordOutStart, IntroMotion.wordOutEnd));
    final logo = IntroMotion.easeOut.transform(IntroMotion.span(t, IntroMotion.logoStart, IntroMotion.logoEnd));
    final dot = IntroMotion.easeOut.transform(IntroMotion.span(t, IntroMotion.dotStart, IntroMotion.dotEnd));
    return Stack(
      alignment: Alignment.center,
      children: [
        if (wordOut < 1)
          Opacity(
            opacity: 1 - wordOut,
            child: Transform.scale(
              scale: 1 - 0.06 * wordOut,
              child: IntroWordmark(edge: IntroMotion.revealEdge(draw)),
            ),
          ),
        if (logo > 0)
          Opacity(
            opacity: logo,
            child: Transform.scale(scale: 0.82 + 0.18 * logo, child: IntroLogo(dot: dot)),
          ),
      ],
    );
  }
}

/// "SubMac" revealed left to right up to [edge] (0..1) through a soft mask, with a thin pen line riding the edge.
class IntroWordmark extends StatelessWidget {
  const IntroWordmark({super.key, required this.edge, this.pen = 1});
  final double edge;
  final double pen;

  @override
  Widget build(BuildContext context) {
    const soft = 0.08;
    // The mask runs a little past both ends so the first and last letters start and finish cleanly.
    final e = -soft + (1 + 2 * soft) * edge;
    return CustomPaint(
      foregroundPainter: _PenPainter(edge: e, opacity: edge <= 0 || edge >= 1 ? 0 : pen),
      child: ShaderMask(
        blendMode: BlendMode.dstIn,
        shaderCallback: (r) => LinearGradient(
          colors: const [IntroMotion.ink, IntroMotion.ink, IntroMotion.inkClear],
          stops: [0, (e - soft).clamp(0.0, 1.0), e.clamp(0.0, 1.0)],
        ).createShader(r),
        child: Padding(
          padding: const EdgeInsets.symmetric(horizontal: 6),
          child: Text(
            IntroMotion.word,
            textDirection: TextDirection.ltr,
            style: Brand.serif(size: IntroMotion.wordSize, color: IntroMotion.ink, height: 1.1),
          ),
        ),
      ),
    );
  }
}

class _PenPainter extends CustomPainter {
  _PenPainter({required this.edge, required this.opacity});
  final double edge;
  final double opacity;

  @override
  void paint(Canvas canvas, Size size) {
    if (opacity <= 0) return;
    final x = size.width * edge.clamp(0.0, 1.0);
    final paint = Paint()
      ..strokeWidth = 2
      ..strokeCap = StrokeCap.round
      ..color = IntroMotion.theme.withValues(alpha: opacity);
    canvas.drawLine(Offset(x, size.height * 0.16), Offset(x, size.height * 0.88), paint);
  }

  @override
  bool shouldRepaint(_PenPainter old) => old.edge != edge || old.opacity != opacity;
}

  final double t;

  @override

import 'dart:async';

import 'package:audioplayers/audioplayers.dart';

/// The intro's sound: one soft rounded pluck per letter of the wordmark, on the haptic ticks, then a warm two-note bloom
/// as the logo resolves and the dot lands (assets/intro/intro_sound.m4a, 3.3 s, about -17.5 LUFS; synthesized for
/// SubMac by tools/intro_sound.py, no samples).
///
/// It never gets in the way:
/// * an ambient session: it mixes with (and on Android briefly ducks) the user's music, never stops it;
/// * silent when the phone is muted: Android plays it as a sonification sound on the system stream, which follows
///   silent and vibrate mode; iOS uses the ambient category, which follows the mute switch;
/// * any failure to load or play is swallowed, so the intro runs on silently;
/// * [stop] cuts it at once (skip).
abstract class IntroSound {
  Future<void> play();
  Future<void> stop();

  /// The real player. Tests pass their own [IntroSound].
  factory IntroSound.device() => _DeviceIntroSound();
}

class _DeviceIntroSound implements IntroSound {
  static const asset = 'intro/intro_sound.m4a'; // under AudioCache's 'assets/' prefix

  static final context = AudioContext(
    android: const AudioContextAndroid(
      contentType: AndroidContentType.sonification,
      usageType: AndroidUsageType.assistanceSonification,
      audioFocus: AndroidAudioFocus.gainTransientMayDuck,
    ),
    iOS: AudioContextIOS(category: AVAudioSessionCategory.ambient),
  );

  AudioPlayer? _player;
  bool _stopped = false;

  @override
  Future<void> play() async {
    try {
      final p = _player ??= AudioPlayer(playerId: 'submac_intro');
      await p.setAudioContext(context);
      await p.setReleaseMode(ReleaseMode.release);
      if (_stopped) return;
      await p.play(AssetSource(asset), volume: 0.9);
      if (_stopped) await p.stop();
      unawaited(p.onPlayerComplete.first.then((_) => _release()).catchError((_) {}));
    } catch (_) {
      // No sound is fine: the intro carries on.
    }
  }

  @override
  Future<void> stop() async {
    _stopped = true;
    try {
      await _player?.stop();
    } catch (_) {}
    await _release();
  }

  Future<void> _release() async {
    final p = _player;
    _player = null;
    try {
      await p?.dispose();
    } catch (_) {}
  }
}

/// What the agent is doing, shown through the brand mark.
enum MarkMode { idle, thinking, working, roaming }

/// The SubMac mark (the serif "S") as a living indicator. No dots (owner rule): motion is carried by the letter
/// and one fine line.
/// - idle: the S breathes slowly;
/// - thinking: the S glows softly while a fine ring opens out around it;
/// - working: a short fine arc turns around the S;
/// - roaming (browsing or using the phone): a longer arc sweeps around the S, a little faster.
class AgentMark extends StatefulWidget {
  const AgentMark({super.key, this.size = 28, this.mode = MarkMode.thinking});

class _AgentMarkState extends State<AgentMark> with SingleTickerProviderStateMixin {
  late final AnimationController _c = AnimationController(vsync: this, duration: _period(widget.mode));

  static Duration _period(MarkMode m) => switch (m) {
        MarkMode.idle => const Duration(milliseconds: 3200),
        MarkMode.thinking => const Duration(milliseconds: 1600),
        MarkMode.working => const Duration(milliseconds: 1100),
        MarkMode.roaming => const Duration(milliseconds: 1800),
      };

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();
    final still = MediaQuery.maybeOf(context)?.disableAnimations ?? false;
    if (still) {
      _c.stop();
      _c.value = 0.5;
    } else if (!_c.isAnimating) {
      _c.repeat();
    }
  }

  @override
  void didUpdateWidget(AgentMark old) {
    super.didUpdateWidget(old);
    if (old.mode != widget.mode) {
      _c.duration = _period(widget.mode);
      if (_c.isAnimating) _c.repeat();
    }
  }

  @override
  void dispose() {
    _c.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final p = Palette.of(context);
    final s = widget.size;
    return SizedBox(
      width: s,
      height: s,
      child: AnimatedBuilder(
        animation: _c,
        builder: (context, _) {
          final t = _c.value;
          final wave = 0.5 - 0.5 * math.cos(2 * math.pi * t); // 0 to 1 and back
          final glow = switch (widget.mode) {
            MarkMode.idle => 0.78 + 0.22 * wave,
            MarkMode.thinking => 0.7 + 0.3 * wave,
            MarkMode.working => 1.0,
            MarkMode.roaming => 0.85 + 0.15 * wave,
          };
          return Stack(
            clipBehavior: Clip.none,
            children: [
              Positioned.fill(child: CustomPaint(painter: _MarkLine(mode: widget.mode, t: t, color: p.text))),
              Positioned(
                left: s * 0.24,
                top: s * 0.02,
                child: Opacity(
                  opacity: glow.clamp(0.0, 1.0),
                  child: Text('S', style: Brand.serif(size: s * 0.86, color: p.text, height: 1.0)),
                ),
              ),
            ],
          );
        },
      ),
    );
  }
}

/// The one fine line around the mark: a ring that opens out (thinking) or an arc that turns (working, roaming).
class _MarkLine extends CustomPainter {
  _MarkLine({required this.mode, required this.t, required this.color});
  final MarkMode mode;
  final double t;
  final Color color;

  @override
  void paint(Canvas canvas, Size size) {
    final c = size.center(Offset.zero);
    final r = size.shortestSide / 2;
    final paint = Paint()
      ..style = PaintingStyle.stroke
      ..strokeCap = StrokeCap.round
      ..strokeWidth = (size.shortestSide / 28).clamp(1.0, 2.4).toDouble();
    switch (mode) {
      case MarkMode.idle:
        return;
      case MarkMode.thinking:
        paint.color = color.withValues(alpha: 0.45 * (1 - t));
        canvas.drawCircle(c, r * (0.75 + 0.35 * t), paint);
      case MarkMode.working:
        paint.color = color.withValues(alpha: 0.7);
        canvas.drawArc(Rect.fromCircle(center: c, radius: r * 0.98), 2 * math.pi * t - math.pi / 2, math.pi * 0.45, false, paint);
      case MarkMode.roaming:
        paint.color = color.withValues(alpha: 0.55);
        canvas.drawArc(Rect.fromCircle(center: c, radius: r * 0.98), 2 * math.pi * t - math.pi / 2, math.pi * 0.9, false, paint);
    }
  }

  @override
  bool shouldRepaint(_MarkLine old) => old.t != t || old.mode != mode || old.color != color;
}
